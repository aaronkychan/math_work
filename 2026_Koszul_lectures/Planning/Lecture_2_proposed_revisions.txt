```tex
Let $A$ be a graded algebra and set $S=A_0$. The degree zero algebra $S$ acts on both sides of $A_1$.
We therefore need the tensor product over an algebra, not only the tensor product over $\Bbbk$.

\begin{definition}\label{def:module-tensor}
Let $M$ be a right $S$-module and let $N$ be a left $S$-module.
The \defn{tensor product} $M\otimes_S N$ is the quotient of $M\otimes_\Bbbk N$ by the subspace generated by
\[
ms\otimes n-m\otimes sn
\qquad (m\in M,\ n\in N,\ s\in S).
\]
\end{definition}

\begin{lemma}[Basic properties of tensor products]\label{lem:tensor-properties}
\begin{enumerate}
\item For a vector space $W$, a bilinear map $b:M\times N\to W$ factors uniquely through $M\otimes_S N$ if and only if $b(ms,n)=b(m,sn)$ for all $m\in M$, $s\in S$, and $n\in N$.
\item For $S$-bimodules $M,N$, the tensor product is an $S$-bimodule with $s(m\otimes n)t=sm\otimes nt$.
\item For $S$-bimodules $U,V,W$, there are bimodule isomorphisms
\[
\begin{aligned}
(U\otimes_S V)\otimes_S W&\xrightarrow{\sim}U\otimes_S(V\otimes_S W),
&(u\otimes v)\otimes w&\longmapsto u\otimes(v\otimes w),\\
S\otimes_S V&\xrightarrow{\sim}V,
&s\otimes v&\longmapsto sv,\\
V\otimes_S S&\xrightarrow{\sim}V,
&v\otimes s&\longmapsto vs.
\end{aligned}
\]
\item Tensor products distribute over direct sums in either variable; for example,
$ (\bigoplus_i M_i)\otimes_S N\cong\bigoplus_i(M_i\otimes_S N)$.
\end{enumerate}
\end{lemma}

\begin{proof}
In (1), Definition~\ref{def:vector-tensor} gives a unique map $M\otimes_\Bbbk N\to W$; it descends to the quotient exactly when it kills the relations in Definition~\ref{def:module-tensor}.
The action in (2) preserves those relations because the two actions on each bimodule commute.
For (3), apply (1) to the displayed maps and their inverses; the unit inverses are $v\mapsto1\otimes v$ and $v\mapsto v\otimes1$.
For (4), apply (1) to $((m_i)_i,n)\mapsto(m_i\otimes n)_i$; the inverse is induced by the inclusions $M_i\to\bigoplus_i M_i$. Both composites fix pure tensors. The argument for the other variable is identical.
\end{proof}
```

```tex
Let $A$ be a graded algebra and set $S=A_0$.
Multiplication on $A_1$ satisfies $(vs)w=v(sw)$ for $v,w\in A_1$ and $s\in S$.
A tensor product over $S$ incorporates this identity.

\begin{definition}\label{def:module-tensor}
For a $\Bbbk$-algebra $S$, a right $S$-module $M$, and a left $S$-module $N$, the \defn{tensor product} $M\otimes_S N$ is the quotient of $M\otimes_\Bbbk N$ by
\[
\operatorname{span}_\Bbbk\{ms\otimes n-m\otimes sn\mid m\in M,\ n\in N,\ s\in S\}.
\]
\end{definition}

Write $m\otimes_S n$ for the image of $m\otimes n$.
No semisimplicity or finiteness assumption on $S,M,N$ is needed for the following properties.

\begin{lemma}[Basic properties of tensor products]\label{lem:tensor-properties}
Let $S$ be a $\Bbbk$-algebra, $M\in\Mod S$, and $N\in\Mod(S^\op)$.
\begin{enumerate}
\item For a vector space $W$ and a bilinear map $b:M\times N\to W$ satisfying $b(ms,n)=b(m,sn)$ for all $m\in M$, $s\in S$, and $n\in N$, there is a unique linear map $\widetilde b$ making the diagram commute:
\[
\xymatrix@C=55pt@R=22pt{
M\times N \ar[r]^{(m,n)\mapsto m\otimes_S n} \ar[dr]_{b}
& M\otimes_S N \ar@{-->}[d]^{\exists!\,\widetilde b}\\
& W.
}
\]
Conversely, every linear map $M\otimes_S N\to W$ defines such a bilinear map.
\item If $M,N$ are $S$-bimodules, then $M\otimes_S N$ is an $S$-bimodule with
$s(m\otimes_S n)t=sm\otimes_S nt$ for $s,t\in S$.
\item For $S$-bimodules $U,V,W$, there are bimodule isomorphisms
\[
\begin{aligned}
(U\otimes_S V)\otimes_S W&\xrightarrow{\sim}U\otimes_S(V\otimes_S W),
&(u\otimes v)\otimes w&\longmapsto u\otimes(v\otimes w),\\
S\otimes_S V&\xrightarrow{\sim}V,
&s\otimes v&\longmapsto sv,\\
V\otimes_S S&\xrightarrow{\sim}V,
&v\otimes s&\longmapsto vs.
\end{aligned}
\]
\item Tensor products distribute over direct sums in either variable. For a family $(M_i)_{i\in I}$ of right $S$-modules, the isomorphism is
\[
\Bigl(\bigoplus_{i\in I}M_i\Bigr)\otimes_S N
\xrightarrow{\sim}\bigoplus_{i\in I}(M_i\otimes_S N),
\qquad (m_i)_i\otimes n\longmapsto(m_i\otimes n)_i.
\]
\end{enumerate}
\end{lemma}

\begin{proof}
For (1), Definition~\ref{def:vector-tensor} gives a unique linear map $M\otimes_\Bbbk N\to W$ sending $m\otimes n$ to $b(m,n)$.
It descends to the quotient precisely when $b(ms,n)-b(m,sn)=0$.
Uniqueness follows because the elements $m\otimes_S n$ span the quotient.

For (2), the outer actions preserve the defining relations: for $a,s,t\in S$,
\[
s(ma\otimes n-m\otimes an)t
=(sm)a\otimes nt-sm\otimes a(nt).
\]
The module axioms follow from those on $M,N$.
In (3), both parenthesisations impose the relations that move an element of $S$ between adjacent factors, so the displayed associativity map and its reverse are well-defined.
Their composites fix every $(u\otimes v)\otimes w$ and $u\otimes(v\otimes w)$.
The unit inverses are $V\to S\otimes_S V$, $v\mapsto1\otimes v$, and $V\to V\otimes_S S$, $v\mapsto v\otimes1$; the tensor relations give the required identities.

For (4), the displayed map has finite support because $(m_i)_i$ does.
It is well-defined by (1).
If $\iota_i:M_i\to\bigoplus_{j\in I}M_j$ is the canonical inclusion, the inverse sends $m\otimes n$ in the $i$th summand to $\iota_i(m)\otimes n$ and extends linearly over the direct sum.
Both composites fix the displayed generators.
The same construction works in the other variable.
\end{proof}

For right module maps $u:M\to M'$ and left module maps $v:N\to N'$, the universal property defines
\[
u\otimes_S v:M\otimes_S N\longrightarrow M'\otimes_S N',
\qquad m\otimes n\longmapsto u(m)\otimes v(n).
\]
These maps satisfy
\[
(u'\otimes v')\circ(u\otimes v)=(u'\circ u)\otimes(v'\circ v),
\qquad
\id_M\otimes\id_N=\id_{M\otimes_S N}
\]
whenever the maps are composable.
Thus, for fixed $N$ and fixed $M$, respectively, tensor product gives functors
\[
(-)\otimes_S N:\Mod S\longrightarrow\Mod\Bbbk,
\qquad
M\otimes_S(-):\Mod(S^\op)\longrightarrow\Mod\Bbbk.
\]
For bimodules and bimodule maps, the same construction preserves the outer actions in Lemma~\ref{lem:tensor-properties}(2).
```

```tex
Tensor duality reverses the order of the factors.

\begin{proposition}\label{prop:tensor-duals}
Let $S$ be finite-dimensional semisimple and let $V,W$ be finite-dimensional $S$-bimodules.
There are bimodule isomorphisms compatible with bimodule maps in $V,W$,
\begin{equation}\label{eq:tensor-duals}
\begin{aligned}
\theta_D &: D(W)\otimes_S V^*\xrightarrow{\sim}D(V\otimes_S W),\\
\theta_r &: W^*\otimes_S V^*\xrightarrow{\sim}(V\otimes_S W)^*,\\
\theta_\ell &: {}^*W\otimes_S {}^*V\xrightarrow{\sim}{}^*(V\otimes_S W),\\
\theta_{\mathrm{en}} &: W^*\otimes_S V^{*_{\mathrm{en}}}
\xrightarrow{\sim}(V\otimes_S W)^{*_{\mathrm{en}}},
\end{aligned}
\end{equation}
defined by
\begin{equation}\label{eq:tensor-evaluations}
\begin{aligned}
\theta_D(\lambda\otimes f)(v\otimes w)&=\lambda(f(v)w),\\
\theta_r(g\otimes f)(v\otimes w)&=g(f(v)w),\\
\theta_\ell(g\otimes f)(v\otimes w)&=f(vg(w)),\\
\theta_{\mathrm{en}}(g\otimes F)(v\otimes w)&=\sum_j g(a_jw)\otimes b_j^\op,
\quad\text{where }F(v)=\sum_j a_j\otimes b_j^\op.
\end{aligned}
\end{equation}
\end{proposition}

\begin{proof}
The formulas in \eqref{eq:tensor-evaluations} respect the tensor relations and bimodule actions.
The maps $\theta_D$ and $\theta_r$ are isomorphisms for $V=S$ as a right module, hence for finite direct sums and direct summands of $S$.
Corollary~\ref{cor:semisimple-summands} applies to $V$.
For $\theta_\ell$, apply the same argument to the left module $W$.
Choose a symmetrising form $\tau$ as in Definition~\ref{def:symmetrising-form}, and use $\gamma_V,\gamma_{V\otimes_S W}$ from \eqref{eq:en-dual-comparison}.
Then
\[
\gamma_{V\otimes_S W}\circ\theta_{\mathrm{en}}
=\theta_r\circ(\id_{W^*}\otimes\gamma_V),
\]
since both sides evaluate $g\otimes F$ at $v\otimes w$ as $\sum_j g(a_jw)\tau(b_j)$.
Proposition~\ref{prop:dual-comparison} makes both $\gamma$ maps isomorphisms, so $\theta_{\mathrm{en}}$ is an isomorphism.
\end{proof}

The maps $\theta_r,\theta_\ell$ are the evaluations needed for quadratic duality \cite[\S2.7]{BGS96}. The maps $\theta_D,\theta_{\mathrm{en}}$ record their compatibility with linear and two-sided duals \cite[\S2.1]{GI20}; their extra formulas may be kept as reference material.

For a fixed symmetrising form $\tau$, let $\alpha_M(f)=\tau\circ f$ as in \eqref{eq:dual-comparison}.
Then tensor evaluation satisfies
\[
\alpha_{V\otimes_S W}\circ\theta_r
=\theta_D\circ(\alpha_W\otimes\id_{V^*}).
\]
Both sides send $g\otimes f$ to $v\otimes w\mapsto\tau(g(f(v)w))$.

\begin{example}
Let $S=\Bbbk^{\oplus r}$ with coordinate idempotents $e_1,\ldots,e_r$, and suppose $V=e_iVe_j$ and $W=e_jWe_\ell$.
By \eqref{eq:bimodule-dual-actions},
\[
V\otimes_S W=e_i(V\otimes_S W)e_\ell,\qquad
W^*\otimes_S V^*=e_\ell(W^*\otimes_S V^*)e_i.
\]
Thus $\theta_r$ in \eqref{eq:tensor-duals} reverses the two outer idempotents.
\end{example}
```

```tex
For an $S$-bimodule $M$, the right and left $S$-duals of Definition~\ref{def:S-duals} are themselves $S$-bimodules, with actions
\begin{equation}\label{eq:tensor-dual-actions}
(sft)(m)=sf(tm),\qquad
(sgt)(m)=g(ms)t
\quad(s,t\in S,\ m\in M).
\end{equation}
Here $f\in M^*$ and $g\in{}^*M$.
The duals are ungraded; the parenthesised stars of Definition~\ref{def:graded-S-duals} denote different, graded constructions.

Tensor duality reverses the order of the factors.

\begin{proposition}\label{prop:tensor-duals}
Let $S$ be finite-dimensional semisimple and let $V,W$ be finite-dimensional $S$-bimodules.
There are bimodule isomorphisms, natural in $V,W$,
\begin{equation}\label{eq:tensor-duals}
\begin{aligned}
\theta_r &: W^*\otimes_S V^*\xrightarrow{\sim}(V\otimes_S W)^*,\\
\theta_\ell &: {}^*W\otimes_S {}^*V\xrightarrow{\sim}{}^*(V\otimes_S W),
\end{aligned}
\end{equation}
defined by
\begin{equation}\label{eq:tensor-evaluations}
\theta_r(g\otimes f)(v\otimes w)=g(f(v)w),
\qquad
\theta_\ell(g\otimes f)(v\otimes w)=f(vg(w)).
\end{equation}
\end{proposition}

\begin{proof}
\emph{Well-definedness.}
For the right duals, right $S$-linearity gives
\[
g(f(vs)w)=g(f(v)sw),\qquad
(gs)(f(v)w)=g(sf(v)w)=g((sf)(v)w).
\]
The first identity respects the relation in $V\otimes_S W$, and the second respects the relation in $W^*\otimes_S V^*$.
The resulting functional is right $S$-linear because $g$ is.
For the left duals, the corresponding identities are
\[
f(vs\,g(w))=f(vg(sw)),\qquad
f(v(gs)(w))=f(vg(w)s)=(sf)(vg(w)).
\]
Left $S$-linearity of $f$ makes the resulting functional left $S$-linear.
The outer actions in \eqref{eq:tensor-dual-actions} give
\[
\begin{aligned}
\theta_r(sg\otimes ft)(v\otimes w)&=s\,g(f(tv)w),\\
\theta_\ell(sg\otimes ft)(v\otimes w)&=f(vg(ws))t,
\end{aligned}
\]
which are the required bimodule identities.

\emph{Bijectivity.}
To check $\theta_r$, forget the left action on $V$.
For $V=S$, identify $S^*\cong S$ by $f\mapsto f(1)$; then $\theta_r$ is $W^*\otimes_S S\to W^*$, $g\otimes s\mapsto gs$, an isomorphism by Lemma~\ref{lem:tensor-properties}(3).
The formula commutes with maps of right $S$-modules in $V$, so it is an isomorphism for finite direct sums and direct summands of $S$.
Corollary~\ref{cor:semisimple-summands} applies to $V$.
For $\theta_\ell$, use $W=S$ as a left module, where the map is $S\otimes_S{}^*V\to{}^*V$, $s\otimes f\mapsto sf$, and apply the same argument to $W$.

\emph{Naturality.}
For bimodule maps $p:V\to V'$ and $q:W\to W'$, the square
\[
\xymatrix@C=48pt@R=22pt{
(W')^*\otimes_S(V')^* \ar[r]^{\theta_r} \ar[d]_{q^*\otimes p^*}
& (V'\otimes_S W')^* \ar[d]^{(p\otimes q)^*}\\
W^*\otimes_S V^* \ar[r]_{\theta_r}
& (V\otimes_S W)^*
}
\]
commutes: both routes send $g\otimes f$, evaluated at $v\otimes w$, to $g(f(p(v))q(w))$.
For $\theta_\ell$, the corresponding two routes give $f(p(v)g(q(w)))$.
\end{proof}

Neither map in \eqref{eq:tensor-duals} requires a symmetrising form.
These canonical identifications are the reason for using $S$-duals in tensor calculations; see \cite[\S2.7]{BGS96}.

\begin{advanced}
For the ungraded $\Bbbk$-linear dual $D$, there is also a natural bimodule isomorphism \cite[\S2.1]{GI20}
\[
\begin{aligned}
\theta_D:D(W)\otimes_S V^*&\xrightarrow{\sim}D(V\otimes_S W),\\
\theta_D(\lambda\otimes f)(v\otimes w)&=\lambda(f(v)w).
\end{aligned}
\]
The actions on $D(W)$ are those in \eqref{eq:bimodule-dual-actions}.
The tensor relations are checked as for $\theta_r$.
For $V=S$, the map is $D(W)\otimes_S S\to D(W)$, $\lambda\otimes s\mapsto\lambda s$; the direct summand argument in Proposition~\ref{prop:tensor-duals} proves bijectivity.

Suppose now that $S$ is equipped with a symmetrising form $\tau:S\to\Bbbk$, as in Definition~\ref{def:symmetrising-form}.
The two-sided dual of Definition~\ref{def:en-dual} satisfies
\begin{equation}\label{eq:tensor-two-sided-dual}
\begin{aligned}
\theta_{\en}:W^*\otimes_S V^{*_{\en}}
&\xrightarrow{\sim}(V\otimes_S W)^{*_{\en}},\\
\theta_{\en}(g\otimes F)(v\otimes w)
&=\sum_j g(a_jw)\otimes b_j^\op,
\qquad F(v)=\sum_j a_j\otimes b_j^\op.
\end{aligned}
\end{equation}
The formula respects the tensor relations and both outer actions.
For example, $F(vs)=F(v)(s\otimes1)$ replaces $a_j$ by $a_js$, while
$(Fs)(v)=(1\otimes s^\op)F(v)$ replaces $b_j$ by $b_js$.
These give, respectively, compatibility with $vs\otimes w=v\otimes sw$ and with the right outer action.
Compatibility with $(gs)\otimes F=g\otimes(sF)$ follows from $(gs)(a_jw)=g(sa_jw)$.
The other actions follow from right $S$-linearity of $g$ and $F(tv)=F(v)(1\otimes t^\op)$.

The comparison maps $\gamma_M$ in \eqref{eq:en-dual-comparison} give
\[
\gamma_{V\otimes_S W}\circ\theta_{\en}
=\theta_r\circ(\id_{W^*}\otimes\gamma_V),
\]
since both sides send $g\otimes F$, evaluated at $v\otimes w$, to $\sum_j g(a_jw)\tau(b_j)$.
Thus $\theta_{\en}$ is an isomorphism by Proposition~\ref{prop:dual-comparison}.
Precomposition shows that $\theta_D$ and $\theta_{\en}$ are natural.
No assumption that $S^\en$ is semisimple is used; compare Remark~\ref{rem:enveloping-semisimple}.

For the maps $\alpha_M(f)=\tau\circ f$ in \eqref{eq:dual-comparison},
\[
\alpha_{V\otimes_S W}\circ\theta_r
=\theta_D\circ(\alpha_W\otimes\id_{V^*}),
\]
because both sides send $g\otimes f$ to the functional $v\otimes w\mapsto\tau(g(f(v)w))$.
\end{advanced}

\begin{example}
Let $S=\Bbbk^{\oplus r}$ with coordinate idempotents $e_1,\ldots,e_r$, and suppose $V=e_iVe_j$ and $W=e_jWe_\ell$.
By \eqref{eq:tensor-dual-actions},
\[
V\otimes_S W=e_i(V\otimes_S W)e_\ell,\qquad
W^*\otimes_S V^*=e_\ell(W^*\otimes_S V^*)e_i.
\]
Thus $\theta_r$ in \eqref{eq:tensor-duals} reverses the two outer idempotents.
\end{example}
```

```tex
\end{lemma}
```

REPLACEMENT CONTENT (exact):
```tex
The decomposition in Definition~\ref{def:total-tensor-grading} follows from Lemma~\ref{lem:tensor-properties}(4), applied over $\Bbbk$:
\[
\Bigl(\bigoplus_iV_i\Bigr)\otimes_\Bbbk
\Bigl(\bigoplus_jW_j\Bigr)
\cong\bigoplus_{i,j}(V_i\otimes_\Bbbk W_j).
\]
An individual tensor has finitely many non-zero components, even if the sum defining a fixed total degree has infinitely many summands.

\begin{lemma}\label{lem:hilbert-tensor}
For locally finite graded vector spaces $V,W$ concentrated in degrees bounded below, $V\otimes_\Bbbk W$ is locally finite and bounded below, and
$h_{V\otimes_\Bbbk W}(t)=h_V(t)h_W(t)$.
\end{lemma}

\begin{proof}
Choose integers $a,b$ with $V_i=0$ for $i<a$ and $W_j=0$ for $j<b$.
For total degree $n$, only $a\leq i\leq n-b$ can contribute, so
\[
\dim_\Bbbk(V\otimes_\Bbbk W)_n
=\sum_{i+j=n}\dim_\Bbbk V_i\dim_\Bbbk W_j
\]
is a finite sum and is zero for $n<a+b$.
The sum is precisely the coefficient of $t^n$ in $h_V(t)h_W(t)$.
\end{proof}
```

REPLACEMENT CONTENT (exact):
```tex
Let $S$ be finite-dimensional semisimple.

\begin{definition}\label{def:orthogonals}
For an $S$-sub-bimodule $U$ of a finite-dimensional $S$-bimodule $M$, its \defn{right orthogonal complement} and \defn{left orthogonal complement} are
\[
\begin{aligned}
U^\perp&:=\{f\in M^*\mid f(u)=0\text{ for all }u\in U\},\\
{}^\perp U&:=\{g\in{}^*M\mid g(u)=0\text{ for all }u\in U\}.
\end{aligned}
\]
\end{definition}

Both orthogonal complements are $S$-sub-bimodules.
Indeed, for $s,t\in S$ and $u\in U$, \eqref{eq:tensor-dual-actions} gives
\[
(sft)(u)=sf(tu)=0\quad(f\in U^\perp),\qquad
(sgt)(u)=g(us)t=0\quad(g\in{}^\perp U),
\]
since $tu,us\in U$.
The orthogonal complements are defined without choosing a symmetrising form.

If $S$ is equipped with a symmetrising form $\tau:S\to\Bbbk$, the comparison maps $\alpha_M,\beta_M$ of \eqref{eq:dual-comparison} satisfy
\begin{equation}\label{eq:orthogonal-comparison}
\alpha_M(U^\perp)=\beta_M({}^\perp U)
=\{\lambda\in D(M)\mid\lambda(u)=0\text{ for all }u\in U\}.
\end{equation}
The forward inclusions follow by composing with $\tau$.
Conversely, $\alpha_M(f)(U)=0$ gives
\[
\tau(f(u)s)=\tau(f(us))=0
\quad\text{for all }s\in S,\ u\in U.
\]
Non-degeneracy of $\tau$ forces $f(u)=0$ for every $u\in U$.
For the left dual, $\tau(sg(u))=\tau(g(su))=0$ gives $g(u)=0$ instead.
Bijectivity of $\alpha_M,\beta_M$ in Proposition~\ref{prop:dual-comparison} proves \eqref{eq:orthogonal-comparison}.

```
