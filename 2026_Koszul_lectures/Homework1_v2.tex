\titledquestion{Graded algebras, duals, and non-graded modules}

\begin{parts}
\part For $A=\Bbbk A_n$ with $n\ge 2$, verify that $A_d=\operatorname{span}_\kk\{e_{ij}\mid j-i=d\}$ defines a
grading on $A$.
\begin{solution}
The $e_{ij}$ ($i\leq j$) form a basis, so $A=\bigoplus_dA_d$, and $1=\sum_ie_{ii}\in A_0$.
If $j-i=d$, $q-p=d'$ and $e_{ij}e_{pq}\neq0$, then $j=p$ and $e_{ij}e_{pq}=e_{iq}$ with
$q-i=(q-p)+(j-i)=d+d'$; hence $A_dA_{d'}\subset A_{d+d'}$.
\end{solution}

\part Compute $\rad(\kk A_n)$ from its
definition as the intersection of all maximal right ideals.
\begin{solution}
Let $J=\operatorname{span}\{e_{ij}\mid i<j\}$; it is a two-sided ideal and $J^n=0$,
since a product of $n$ matrices in $J$ vanishes.
\emph{$J\subset\rad A$.} Let $\mathfrak m$ be a maximal right ideal; $L=A/\mathfrak m$ is simple.
Then $LJ$ is a submodule, so $LJ\in\{0,L\}$, and $LJ=L$ would give $L=LJ^n=0$. Hence $LJ=0$,
i.e.\ $J=1\cdot J\subset\mathfrak m$.
\emph{$\rad A\subset J$.} For each $i$, $X\mapsto X_{ii}$ is a surjective algebra homomorphism $A\to\kk$,
so $\mathfrak m_i=\{X\mid X_{ii}=0\}$ is a right ideal with $A/\mathfrak m_i\cong\kk$ simple,
i.e.\ a maximal right ideal. Then $\bigcap_i\mathfrak m_i=\{X\mid\text{all }X_{ii}=0\}=J$.
Hence $\rad A=J$.
\end{solution}

\part Let $A=\Bbbk A_3$, $M=e_{22}A$, and
consider $\bbD(M)_d=\Hom_\kk(M_{-d},\kk)$ with $(af)(m)=f(ma)$.
\begin{subparts}
\subpart Find a homogeneous basis of $\bbD(M)$, its degrees, and all non-zero
actions of matrix units on it.
\begin{solution}
$M=\kk e_{22}\oplus\kk e_{23}$ with degrees $0,1$. Let $\varepsilon_{22},\varepsilon_{23}$ be the dual basis;
$\deg\varepsilon_{22}=0$, $\deg\varepsilon_{23}=-1$. Since $(af)(m)=f(ma)$ and
$e_{ij}e_{pq}=\delta_{jp}e_{iq}$, the only non-zero actions are
$e_{22}\varepsilon_{22}=\varepsilon_{22}$, $e_{33}\varepsilon_{23}=\varepsilon_{23}$,
$e_{23}\varepsilon_{23}=\varepsilon_{22}$.
(E.g.\ $(e_{23}\varepsilon_{23})(e_{22})=\varepsilon_{23}(e_{22}e_{23})=1$, while $e_{11},e_{12},e_{13}$
act by $0$ because $e_{22}e_{1q}=e_{23}e_{1q}=0$.)
\end{solution}
\subpart Show that $\Bbbk e_{13}\subset Ae_{33}$ is a graded left submodule and
that $\bbD(M)\cong(Ae_{33}/\Bbbk e_{13})(1)$ as graded left $A$-modules.
\begin{solution}
$Ae_{33}=\operatorname{span}\{e_{13},e_{23},e_{33}\}$ and $e_{ij}e_{13}=\delta_{j1}e_{i3}$, so
$\Bbbk e_{13}$ is a (homogeneous) submodule. In $N=Ae_{33}/\Bbbk e_{13}$ the classes $\bar e_{33},\bar e_{23}$
have degrees $0,1$ and the non-zero actions are $e_{33}\bar e_{33}=\bar e_{33}$,
$e_{23}\bar e_{33}=\bar e_{23}$, $e_{22}\bar e_{23}=\bar e_{23}$. In $N(1)$ the degrees are $-1,0$.
The linear map $\varepsilon_{23}\mapsto\bar e_{33}$, $\varepsilon_{22}\mapsto\bar e_{23}$ is bijective, preserves
degrees, and matches the non-zero actions of part (i); as matrix units span $A$, it is $A$-linear.
\end{solution}
\subpart Deduce that $\bbD(M)$ is not isomorphic to any
$(Ae_{jj})(r)$, $j\in\{1,2,3\}$, $r\in\mathbb Z$.
\begin{solution}
$\dim\bbD(M)=2$ and $\dim Ae_{jj}=j$, so only $j=2$ is possible. But $e_{33}$ acts by $0$ on $Ae_{22}=\operatorname{span}\{e_{12},e_{22}\}$
while $e_{33}\varepsilon_{23}=\varepsilon_{23}\neq0$; an isomorphism commutes with $e_{33}$.
\end{solution}
\end{subparts}


\part \textbf{Modules that are not graded.}
Let $A$ be the algebra of matrices
\[
  X=\begin{pmatrix}a&u\alpha+v\beta\\0&b\end{pmatrix}\qquad(a,b,u,v\in\kk),
\]
with the usual matrix product, where scalars act on the space $\kk\alpha\oplus\kk\beta$ in the obvious way:
\[
\begin{pmatrix}a&p\\0&b\end{pmatrix}\begin{pmatrix}a'&p'\\0&b'\end{pmatrix}
=\begin{pmatrix}aa'&ap'+pb'\\0&bb'\end{pmatrix}
\qquad(p,p'\in\kk\alpha\oplus\kk\beta).
\]
Grade $A$ by $\deg\alpha=1$, $\deg\beta=0$, with diagonal entries in degree $0$.
For $(\lambda,\mu)\in\kk^2$ let $M(\lambda,\mu)=\kk^2$ (row vectors) with right action
\[
  (x_1,x_2)\cdot X=\bigl(ax_1,\ (\lambda u+\mu v)x_1+bx_2\bigr).
\]
\begin{subparts}
\subpart Verify that $A$ is a graded algebra and that $M(\lambda,\mu)$ is a right $A$-module for every $(\lambda,\mu)\in\Bbbk^2$.
\begin{solution}
In $ap'+pb'$ the degree of $p$ or $p'$ is unchanged by the scalars $a,b,a',b'$
(all of degree $0$), so $A_dA_e\subset A_{d+e}$; $1\in A_0$.
Put $\phi(u\alpha+v\beta)=\lambda u+\mu v$ and $\rho(X)=\begin{pmatrix}a&\phi(p)\\0&b\end{pmatrix}$.
Then $\phi(ap'+pb')=a\phi(p')+\phi(p)b'$ gives $\rho(XX')=\rho(X)\rho(X')$ and $\rho(1)=I$,
and $x\cdot X=x\rho(X)$ (row vector times matrix); hence $(x\cdot X)\cdot X'=x\cdot(XX')$.
\end{solution}
\subpart Prove that $M(\lambda,\mu)$ admits a grading if and only if $\lambda\mu=0$.
\begin{solution}
Let $e_1=\mathrm{diag}(1,0)$, $e_2=\mathrm{diag}(0,1)\in A_0$, and $m_1=(1,0)$, $m_2=(0,1)$,
so $Me_i=\kk m_i$.
($\Rightarrow$) For a grading, $Me_i=\bigoplus_d(M_de_i)$ is a graded subspace; being one-dimensional it is
spanned by a homogeneous vector, so $m_i\in M_{d_i}$. Now $m_1\alpha=\lambda m_2\in M_{d_1+1}$ and
$m_1\beta=\mu m_2\in M_{d_1}$. If $\lambda\neq0$ then $d_2=d_1+1$; if $\mu\neq0$ then $d_2=d_1$; so $\lambda\mu\neq0$ is impossible.
($\Leftarrow$) If $\mu=0$ take $d_1=0,d_2=1$; if $\mu\neq0$ (so $\lambda=0$) take $d_1=d_2=0$.
Since $m_2X=bm_2$ and $m_1X=am_1+(\lambda u+\mu v)m_2$, the condition $M_dA_e\subset M_{d+e}$ holds.
\end{solution}
\end{subparts}

\part \textbf{Independence of the symmetrising form.}
Let $A=\kk A_2$, $S=\kk e_{11}\oplus\kk e_{22}$ and $M=e_{11}A$ (inherited gradings).
For $c=(c_1,c_2)\in(\kk\setminus\{0\})^2$, consider the symmetrising form $$\tau_c(ae_{11}+be_{22})=c_1a+c_2b.$$ 

Recall that we have the natural isomorphism $\alpha_c:(-)^{(*)}\xrightarrow{\sim} \bbD$ associated to $\tau_c$ given by $f\mapsto\tau_c\circ f$, which induces a left $A$-action $\cdot_c$ on $M^{(*)}$ induced by $\alpha_c$ is given by  \[a\cdot_cf=\alpha_c^{-1}\bigl(a\,\alpha_c(f)\bigr),\qquad\text{where }(a\lambda)(m)=\lambda(ma) \text{ for $\lambda\in \bbD M$}.
\]
Let
$$\varphi_0(ae_{11}+be_{12})=ae_{11}, \qquad   \varphi_1(ae_{11}+be_{12})=be_{22}.$$ \begin{subparts}
\subpart Check that $\varphi_0,\varphi_1$ forms a homogeneous basis of $M^{\grstar}$ (specify their degrees).
Compute $e_{11}\cdot_c\varphi_i$, $e_{22}\cdot_c\varphi_i$, $e_{12}\cdot_c\varphi_i$ for $i\in\{0,1\}$. Prove that
$e\cdot_c\varphi_i=e\cdot_{c'}\varphi_i$ for all $e\in\{e_{11},e_{22},e_{12}\}$ and $i\in\{0,1\}$ if and only if $(c_1',c_2')=\rho(c_1,c_2)$ for some $\rho\in\kk\setminus\{0\}$.
\begin{solution}
For $m=xe_{11}+ye_{12}$: $\alpha_c(\varphi_0)(m)=c_1x$, $\alpha_c(\varphi_1)(m)=c_2y$, and
$me_{11}=xe_{11}$, $me_{22}=ye_{12}$, $me_{12}=xe_{12}$. Since $(a\lambda)(m)=\lambda(ma)$:
$e_{11}\cdot\varphi_0=\varphi_0$, $e_{22}\cdot\varphi_1=\varphi_1$, and
$\alpha_c(e_{12}\cdot\varphi_1)(m)=\alpha_c(\varphi_1)(xe_{12})=c_2x=\tfrac{c_2}{c_1}\alpha_c(\varphi_0)(m)$, i.e.\
$e_{12}\cdot_c\varphi_1=\tfrac{c_2}{c_1}\varphi_0$; all other products vanish
(e.g.\ $\alpha_c(\varphi_0)(me_{22})=0$, $\alpha_c(\varphi_1)(me_{11})=0$, $\alpha_c(\varphi_0)(me_{12})=0$).
Only the last depends on $c$, so the actions agree iff $c_2/c_1=c_2'/c_1'$, i.e.\ iff $c_1'/c_1=c_2'/c_2=:\rho$.
\end{solution}
\subpart More generally, let $A$ be a locally finite-dimensional graded algebra with semisimple $A_0=S$, and
$\tau,\tau'$ be symmetrising forms on $S$ with associated natural isomorphisms $\alpha_\tau$ and left $A$-action $\cdot_\tau$.  
Prove that there is an invertible $z\in S$ with $\tau'(s)=\tau(zs)$ for all $s$, and that
$$\gamma:(M^{\grstar},\cdot_{\tau'})\to (M^{\grstar},\cdot_{\tau}), \quad  (\gamma f)(m)=z\,f(m),$$
is an isomorphism of graded left $A$-modules.
Deduce that in (i) the two modules $(M^{\grstar},\cdot_c)$ and $(M^{\grstar},\cdot_{c'})$ are isomorphic for all $c,c'$.

\emph{Hint.} Compare $\alpha_{\tau'}$ with $\alpha_\tau$.
\begin{solution}
$z\mapsto\tau(z\,\cdot)$ is injective by non-degeneracy of $\tau$, hence bijective $S\to\bbD(S)$ by dimension count; this gives $z$.
If $zs=0$ then $\tau'(st)=\tau(zst)=0$ for all $t$, so $s=0$ by non-degeneracy of $\tau'$.
Thus $s\mapsto zs$ is injective, hence bijective; so $zs_0=1$ for some $s_0$, and in a finite-dimensional algebra a one-sided inverse is two-sided.
Next, $\gamma f$ is right $S$-linear ($zf(ms)=zf(m)s$), $\gamma$ preserves degrees and is invertible (inverse $f\mapsto z^{-1}f$).
Since $\alpha_{\tau'}(f)=\tau(zf(\cdot))=\alpha_\tau(\gamma f)$, we get $\alpha_{\tau'}=\alpha_\tau\circ\gamma$ and
$a\cdot_{\tau'}f=\gamma^{-1}\alpha_\tau^{-1}(a\,\alpha_\tau(\gamma f))=\gamma^{-1}(a\cdot_\tau\gamma f)$, i.e.\ $\gamma(a\cdot_{\tau'}f)=a\cdot_\tau\gamma(f)$.
For (i) take $z=\frac{c_1'}{c_1}e_{11}+\frac{c_2'}{c_2}e_{22}$: $\tau_{c'}=\tau_c(z\,\cdot)$, so $(M^{\grstar},\cdot_{c'})\cong(M^{\grstar},\cdot_c)$ although $\cdot_c\neq\cdot_{c'}$ unless $c'\propto c$.
\end{solution}
\end{subparts}
\end{parts}

\titledquestion{$S$-duals and quadratic duals}
Let $S$ be a finite-dimensional semisimple $\kk$-algebra. 
Let $V$ be a finite-dimensional right $S$-module and $W$ a finite-dimensional
$S$-bimodule, and define
\[
  \theta_r:W^*\otimes_SV^*\longrightarrow(V\otimes_SW)^*,\qquad
  \theta_r(g\otimes f)(v\otimes w)=g(f(v)w).
\]
\begin{parts}
\part Prove that $\theta_r$ is well defined $S$-module homomorphism.
\begin{solution}
For fixed $g,f$, $\beta(v,w)=g(f(v)w)$ is bilinear, and
$\beta(vs,w)=g(f(v)sw)=\beta(v,sw)$ because $f$ is right $S$-linear; so $\beta$ induces a linear map $V\otimes_SW\to S$.
It is right $S$-linear: $\beta(v,ws)=g(f(v)ws)=g(f(v)w)s$ as $g$ is. The formula is bilinear in $(g,f)$, and
$(gs)(f(v)w)=g(sf(v)w)=g((sf)(v)w)$ shows invariance under $gs\otimes f\leftrightarrow g\otimes sf$.
For a bimodule $V$ and the actions $(sft)(v)=sf(tv)$,
\[
\theta_r(sg\otimes ft)(v\otimes w)=s\,g(f(tv)w)=\bigl(s\,\theta_r(g\otimes f)\,t\bigr)(v\otimes w).
\]
\end{solution}

\part Prove that $\theta_r$ is bijective.
\begin{solution}
\emph{$V=S$:} $S^*\cong S$ via $f\mapsto f(1)$ (as $f(s)=f(1)s$), and $S\otimes_SW\cong W$. Under these identifications
$\theta_r(g\otimes f)(s\otimes w)=g(f(1)sw)=(g\,f(1))(sw)$, so $\theta_r$ becomes $W^*\otimes_SS\to W^*$, $g\otimes s\mapsto gs$, an isomorphism.
\emph{$V=S^{\oplus n}$:} both sides turn finite direct sums in $V$ into direct sums, and $\theta_r$ is compatible with the inclusions and projections
by naturality (part (3), applied to right $S$-linear maps); so $\theta_r^{V_1\oplus V_2}=\theta_r^{V_1}\oplus\theta_r^{V_2}$.
\emph{General $V$:} $V\oplus V'\cong S^{\oplus n}$ as right modules ($S$ semisimple), so $\theta_r^V\oplus\theta_r^{V'}$ is bijective, hence so is $\theta_r^V$.
\end{solution}

\part Let $p:V\to V'$ be right $S$-linear and $q:W\to W'$ a bimodule map.
Prove
\[
  (p\otimes q)^*\circ\theta_r^{V',W'}=\theta_r^{V,W}\circ(q^*\otimes p^*).
\]
\begin{solution}
For $g\in(W')^*$, $f\in(V')^*$:
$\bigl((p\otimes q)^*\theta_r(g\otimes f)\bigr)(v\otimes w)=g(f(p(v))\,q(w))$, and
$\theta_r(q^*g\otimes p^*f)(v\otimes w)=g\bigl(q(f(p(v))\,w)\bigr)=g(f(p(v))\,q(w))$ since $q$ is left $S$-linear.
\end{solution}

\part Let
\[
  S=\kk^{\oplus3},\qquad V=\kk a,\quad W=\kk c,\qquad a\in e_1Ve_2,\quad c\in e_2We_3.
\]
Compute $W^*\otimes_SV^*$, $\ V^*\otimes_SW^*$, $\ (V\otimes_SW)^*$ and describe $\theta_r$.
\begin{solution}
$V^*=\kk a^*$ with $a^*(a)=e_2$ (right linearity: $a=ae_2$), and $e_2a^*e_1=a^*$; $W^*=\kk c^*$ with $c^*(c)=e_3$, $e_3c^*e_2=c^*$.
$V\otimes_SW=\kk\,a\otimes c\neq0$ (as $a=ae_2$, $c=e_2c$), so $(V\otimes_SW)^*$ is $1$-dimensional.
$W^*\otimes_SV^*=\kk\,c^*\otimes a^*\neq0$ since $c^*=c^*e_2$, $a^*=e_2a^*$, and
$\theta_r(c^*\otimes a^*)(a\otimes c)=c^*(a^*(a)c)=c^*(e_2c)=e_3\neq0$: $\theta_r$ is an isomorphism.
Finally $a^*\otimes c^*=a^*e_1\otimes c^*=a^*\otimes e_1c^*=a^*\otimes e_1e_3c^*=0$, so $V^*\otimes_SW^*=0$.
\end{solution}

\part Let
\[
  S=\kk[x]/(x^2),\qquad V=W=S/(x).
\]
Show that
\[
  \dim_\kk\bigl(W^*\otimes_SV^*\bigr)=\dim_\kk(V\otimes_SW)^*=1,\qquad\theta_r=0.
\]
At which step of your proof of (2) does the argument fail?
\begin{solution}
Write $\kk=S/(x)$. $\kk^*=\Hom_S(\kk,S)$: $f(\bar1)=u$ requires $ux=0$, so $u\in\kk x$, and $\kk^*\cong\kk$ with $x$ acting by $0$ on both sides.
So $W^*\otimes_SV^*\cong\kk\otimes_S\kk=\kk$, and $(V\otimes_SW)^*=\kk^*\cong\kk$. But $\theta_r(g\otimes f)(\bar1\otimes\bar1)=g(f(\bar1)\bar1)=g(cx\cdot\bar1)=g(0)=0$.
So $\theta_r=0$, neither injective nor surjective. The argument fails in the last step of (2): $V=\kk$ is not a direct summand of any $S^{\oplus n}$
(it is not free), since $S$ is not semisimple.
\end{solution}

\part Let $S=\bigoplus_{i=1}^4\kk e_i$ be a semisimple algebra with coordinate idempotent $e_i$ for $1\le i\le 4$ and let $V=\kk a\oplus\kk b\oplus\kk c\oplus\kk d$ be the $S$-bimodule with
\[
  a\in e_1Ve_2,\qquad b\in e_2Ve_4,\qquad c\in e_1Ve_3,\qquad d\in e_3Ve_4,
\]
and $e_iVe_j=0$ for all other $(i,j)$. For $\lambda\in\kk$ put
\[
  A_\lambda=T_S(V)/(ab-\lambda cd).
\]
\begin{subparts}
\subpart Using $\theta_r$, find $R^\perp$ and a presentation of $A_\lambda^!$, for every $\lambda\in\kk$.

\emph{Hint.} Evaluate $\mu\,b^*a^*+\nu\,d^*c^*$ on $ab-\lambda cd$.
\begin{solution}
$V^*$ has basis $a^*\in e_2V^*e_1$, $b^*\in e_4V^*e_2$, $c^*\in e_3V^*e_1$, $d^*\in e_4V^*e_3$, and the composable words of length $2$ are $b^*a^*$, $d^*c^*$.
$\theta_r(b^*\otimes a^*)(a\otimes b)=b^*(e_2b)=e_4$ and it vanishes on $c\otimes d$; $\theta_r(d^*\otimes c^*)(c\otimes d)=e_4$ and it vanishes on $a\otimes b$.
So
\[
  \theta_r(\mu b^*a^*+\nu d^*c^*)(ab-\lambda cd)=(\mu-\lambda\nu)e_4,
\]
giving $R^\perp=\kk(\lambda b^*a^*+d^*c^*)$ and $A_\lambda^!=T_S(V^*)/(\lambda b^*a^*+d^*c^*)$.
\end{solution}
\subpart Dualise once more: compute ${}^!(A_\lambda^!)$ and recover the relation $ab-\lambda cd$.
\begin{solution}
The element $\alpha\,ab+\beta\,cd$ annihilates $\lambda b^*a^*+d^*c^*$ under the pairing of (i) iff $\alpha\lambda+\beta=0$, so the orthogonal of $R^\perp$ is $\kk(ab-\lambda cd)$.
\end{solution}
\subpart Prove that $A_\lambda\cong A_\mu$ as graded algebras if and only if $\lambda,\mu$ are either both zero or both non-zero.

\emph{Hint.} A graded isomorphism over $S$ sends each of $a,b,c,d$ to a scalar multiple of itself.
\begin{solution}
A graded isomorphism $\varphi:A_\lambda\to A_\mu$ over $S$ fixes each $e_i$, so $\varphi_1$ preserves each $e_iVe_j$ (all at most one-dimensional):
\[
  \varphi(a)=\alpha a,\quad\varphi(b)=\beta b,\quad\varphi(c)=\gamma c,\quad\varphi(d)=\delta d,\qquad\alpha,\beta,\gamma,\delta\in\kk^\times.
\]
The induced map on $T_S(V)$ sends $ab-\lambda cd$ to $\alpha\beta\,ab-\lambda\gamma\delta\,cd$, which must lie in $\kk(ab-\mu cd)$:
\[
  \mu\,\alpha\beta=\lambda\,\gamma\delta.
\]
Hence $\lambda=0\Leftrightarrow\mu=0$. Conversely if $\lambda,\mu\neq0$ take $\alpha=\beta=\gamma=1$, $\delta=\mu/\lambda$ (the inverse is of the same form); for $\lambda=\mu=0$ take the identity.
\end{solution}
\end{subparts}
\end{parts}

\titledquestion{Quadratic presentations and Hilbert series}
\begin{parts}
\part For $n\geq3$ and $Z\subset\{1,\dots,n-2\}$ let
\[
  A_Z=\kk A_n/(e_{k,k+2}\mid k\in Z)
\]
with the quotient grading.
\begin{subparts}
\subpart Show that
\[
  \{e_{ii}\}_{i}\ \cup\ \{e_{ij}\mid i<j,\ \nexists\,k\in Z:\ i\leq k\leq j-2\}
\]
is a basis of $A_Z$.
\begin{solution}
Let $I=\operatorname{span}\{e_{ij}\mid\exists k\in Z:\ i\leq k,\ k+2\leq j\}$. Since $e_{ij}=e_{ik}e_{k,k+2}e_{k+2,j}$, $I$ lies in $(e_{k,k+2}:k\in Z)$;
conversely $I$ is a two-sided ideal containing the $e_{k,k+2}$ (a product $e_{pq}e_{ij}e_{rs}\neq0$ has $p\leq i\leq k$ and $k+2\leq j\leq s$).
So $I=(e_{k,k+2}:k\in Z)$ and the remaining matrix units form a basis of the quotient.
\end{solution}
\subpart Write $A_Z=T_S(V)/(R)$ with explicit $S$, $V$, and $R\subset V\otimes_SV$.
\begin{solution}
\[
  S=\bigoplus_{i=1}^n\kk e_{ii},\qquad V=\bigoplus_{i<n}\kk e_{i,i+1},\qquad R=\operatorname{span}\{e_{k,k+1}\otimes e_{k+1,k+2}\mid k\in Z\}.
\]
Indeed $T_S(V)\cong\kk A_n$ (Lecture 2), and under this
isomorphism $(R)$ goes to the ideal generated by the products $e_{k,k+1}e_{k+1,k+2}=e_{k,k+2}$.
\end{solution}
\end{subparts}

\part Let $S=\kk e_1\oplus\kk e_2$ and $V=\kk a\oplus\kk b$ with
\[
  a\in e_1Ve_2,\qquad b\in e_2Ve_1,
\]
and $e_iVe_j=0$ otherwise.
\begin{subparts}
\subpart Find a basis of $T_S(V)_d$ for all $d\geq0$, and compute $h_{T_S(V)}(t)$.

\emph{Hint.} Count alternating words in each degree, including degree $0$.
\begin{solution}
Positive-degree basis words of $T_S(V)$ are the composable (alternating) words. For each $d\geq1$ there are exactly two:
$aba\cdots$ ($d$ letters, from $e_1$) and $bab\cdots$ (from $e_2$); for $d=0$ the basis is $e_1,e_2$.
The word $aba\cdots$ lies in $e_1T_de_1$ if $d$ is even and in $e_1T_de_2$ if $d$ is odd; $bab\cdots$ lies in $e_2T_de_2$ ($d$ even), $e_2T_de_1$ ($d$ odd). So
\[
h_{T_S(V)}(t)=\frac1{1-t^2}\begin{pmatrix}1&t\\t&1\end{pmatrix}.
\]
\end{solution}
\subpart Find a basis of $A=T_S(V)/(ab)$ and show that $\dim_\kk A<\infty$.
\begin{solution}
The ideal $(ab)$ is spanned by words containing $ab$; every alternating word of length $\geq3$ contains $ab$ ($aba$, $bab$), so a basis is
\[
  e_1,\ e_2,\ a,\ b,\ ba\qquad(ba\in e_2Ae_2),
\]
and $\dim_\kk A=5$.
\end{solution}
\end{subparts}


\part Let
\[
  B=T_\kk(\kk x\oplus\kk y)/(x^2),\qquad\deg x=\deg y=1,\qquad a_d:=\dim_\kk B_d.
\]
\begin{subparts}
\subpart Find a basis of $B_d$ and compute $a_0,\dots,a_6$.
\begin{solution}
A basis of $B_d$ is the set of words in $x,y$ of length $d$ containing no $xx$ (the ideal $(x^2)$ is spanned by words containing $xx$).
\[
  (a_0,\dots,a_6)=(1,2,3,5,8,13,21).
\]
\end{solution}
\subpart Prove that
\[
  a_d=a_{d-1}+a_{d-2}\quad(d\geq2),\qquad h_B(t)=\frac{1+t}{1-t-t^2}.
\]

\emph{Hint.} Sort the words by their first letter.
\begin{solution}
For $d\geq2$: words starting with $y$ are $y\cdot w$ with $w$ admissible of length $d-1$ ($a_{d-1}$ choices); words starting with $x$ must continue with $y$, and are $xy\cdot w$ with $w$ admissible of length $d-2$ ($a_{d-2}$).
So $(1-t-t^2)h_B=a_0+(a_1-a_0)t=1+t$.
\end{solution}
\subpart Compute $B^!$ using $\theta_r$ (with $X=x^*$, $Y=y^*$), find a homogeneous basis of $B^!$ and $h_{B^!}(t)$, and verify
\[
  h_B(t)\,h_{B^!}(-t)=1.
\]
\begin{solution}
For $F=\alpha XX+\beta XY+\gamma YX+\delta YY$ one has $\theta_r(F)(x\otimes x)=\alpha$, so $R^\perp=\operatorname{span}\{XY,YX,YY\}$ and $B^!=\kk\langle X,Y\rangle/(XY,YX,Y^2)$.
Any word of length $\geq2$ containing $Y$ contains one of $XY,YX,YY$; so $\{X^n\mid n\geq0\}\cup\{Y\}$ spans, and it is a basis because the ideal is spanned by words containing $XY$, $YX$ or $YY$, none of which occurs in $X^n$ or $Y$.
Thus $h_{B^!}(t)=\frac1{1-t}+t$ and
\[
  h_B(t)h_{B^!}(-t)=\frac{1+t}{1-t-t^2}\Bigl(\frac1{1+t}-t\Bigr)=\frac{1+t}{1-t-t^2}\cdot\frac{1-t-t^2}{1+t}=1.
\]
\end{solution}
\end{subparts}
\end{parts}

\titledquestion{Computing quadratic duals and Hilbert series}
\begin{parts}
\part Let $S=\bigoplus_{i=1}^r\kk e_i$, let $V$ be a finite-dimensional $S$-bimodule in degree $1$, and put
\[
  C_{ij}=\dim_\kk e_iVe_j,\qquad A=T_S(V)/(V\otimes_SV)=S\oplus V.
\]
\begin{subparts}
\subpart Prove that
\[
  A^!=T_S(V^*),\qquad h_A(t)=I+tC,\qquad h_{A^!}(t)=(I-tC^{\mathsf T})^{-1}.
\]

\emph{Hint.} Use $V\otimes_SV$ as the relation space and $h_{T_S(V)}(t)=(I-tC)^{-1}$ (Lecture~2).
\begin{solution}
The relation space is $R=V\otimes_SV$; since $\theta_r$ identifies $V^*\otimes_SV^*$ with $(V\otimes_SV)^*$, $R^\perp=0$ and $A^!=T_S(V^*)$.
$e_iA_0=L_i$ and $e_iA_1=e_iV$ give $h_A=I+tC$. For $V^*$, $\dim e_iV^*e_j=\dim e_jVe_i$, so its matrix is $C^{\mathsf T}$ and $h_{A^!}=(I-tC^{\mathsf T})^{-1}$.
\end{solution}
\subpart Deduce that
\[
  h_A(t)\,h_{A^!}(-t)^{\mathsf T}=I.
\]
\begin{solution}
$h_{A^!}(-t)^{\mathsf T}=\bigl((I+tC^{\mathsf T})^{-1}\bigr)^{\mathsf T}=(I+tC)^{-1}$, so $h_A(t)h_{A^!}(-t)^{\mathsf T}=I$.
\end{solution}
\end{subparts}

\part For $q\in\kk$ let
\[
  C_q=\kk\langle x,y\rangle/(x^2,\ y^2,\ xy+qyx),\qquad\deg x=\deg y=1.
\]
\begin{subparts}
\subpart Find a homogeneous basis of $C_q$ and compute all products of two basis elements, treating $q=0$ and $q\neq0$. Justify linear independence of your basis.
\begin{solution}
The degree-$2$ relation space $R=\operatorname{span}\{x\otimes x,\ y\otimes y,\ x\otimes y+qy\otimes x\}$ is $3$-dimensional (the three tensors involve different basis tensors), so $\dim(C_q)_2=4-3=1$.
Modulo $R$: $xx\equiv yy\equiv0$ and $xy\equiv-q\,yx$, so $(C_q)_2$ is spanned by the class of $yx$, which is therefore non-zero. In degree $3$, a word contains $xx$ or $yy$ or equals $xyx$ or $yxy$; and
$xyx\equiv-qyxx\equiv0$, $yxy\equiv-qyyx\equiv0$; hence $(C_q)_d=0$ for $d\geq3$. Thus $\{1,x,y,yx\}$ is a basis (dimensions $1,2,1$), and
$h_{C_q}(t)=1+2t+t^2$ for every $q$. Products: $x\cdot x=y\cdot y=0$, $x\cdot y=-q\,yx$, $y\cdot x=yx$, and every product with $yx$ in positive degree is $0$ (degree $\geq3$).
\end{solution}
\subpart For $u=\alpha x+\beta y$ compute $u^2$, and determine
\[
  \{(\alpha,\beta)\in\kk^2\mid u^2=0\}\qquad\text{for each }q,
\]
and the $q$ for which every $u\in(C_q)_1$ satisfies $u^2=0$.
\begin{solution}
$u^2=\alpha^2x^2+\alpha\beta(xy+yx)+\beta^2y^2=\alpha\beta(1-q)\,yx$, and $yx\neq0$. So $u^2=0$ iff $\alpha\beta(1-q)=0$: for $q\neq1$, iff $\alpha=0$ or $\beta=0$
(the two lines $\kk x,\kk y$); for $q=1$, for all $(\alpha,\beta)$. Every element squares to zero iff $q=1$.
\end{solution}
\subpart Prove that
\[
  C_q\cong C_{q'}\ \text{ as graded algebras}\quad\Longleftrightarrow\quad q=q'\ \text{ or }\ qq'=1.
\]
Deduce that $C_1\not\cong C_{-1}$ if $\operatorname{char}\kk\neq2$ (even ungraded), and say what changes if $\operatorname{char}\kk=2$.

\emph{Hint.} A graded isomorphism is determined by a linear bijection of the degree-$1$ parts, and preserves $\{\kk u\mid u^2=0\}$.
\begin{solution}
($\Leftarrow$) If $qq'=1$ then $q\neq0$ and $x\mapsto y$, $y\mapsto x$ sends $x^2,y^2$ to $y^2,x^2$ and $xy+qyx\mapsto yx+qxy=q(xy+q'yx)$, so it induces a graded isomorphism $C_q\to C_{q'}$ (inverse of the same form).
($\Rightarrow$) Let $\varphi:C_q\to C_{q'}$ be a graded isomorphism; $\varphi$ is determined by the bijection $\varphi_1$ on degree $1$ and preserves $u^2=0$.
If $q=1$, every $u$ squares to zero in $C_1$, hence in $C_{q'}$, so $q'=1$ by (ii); symmetrically $q'=1\Rightarrow q=1$. If $q,q'\neq1$, then $\varphi_1$ permutes the two lines $\kk x,\kk y$:
either $\varphi(x)=\lambda x$, $\varphi(y)=\mu y$, and $0=\varphi(xy+qyx)=\lambda\mu(q-q')yx$ gives $q=q'$; or $\varphi(x)=\lambda y$, $\varphi(y)=\mu x$, and
$0=\lambda\mu(yx+qxy)=\lambda\mu(1-qq')yx$ gives $qq'=1$ (here $\lambda\mu\neq0$ and $yx\neq0$).
Now if $\operatorname{char}\kk\neq2$, then $1\neq-1$ and $1\cdot(-1)\neq1$, so $C_1\not\cong C_{-1}$ as graded algebras; even ungraded, $C_{-1}$ is commutative ($xy=yx$) whereas $C_1$ is not ($xy=-yx\neq yx$ as $2yx\neq0$).
In characteristic $2$, $1=-1$, so $C_1=C_{-1}$.
\end{solution}
\subpart Compute a quadratic presentation of $C_q^!$ using $\theta_r$ (with $X=x^*$, $Y=y^*$), and $h_{C_q^!}(t)$. Verify
\[
  h_{C_q}(t)\,h_{C_q^!}(-t)=1,
\]
and compare $C_0^!$ with $C_1^!$.

\emph{Hint.} Evaluate a general combination of $X^2,XY,YX,Y^2$ on $x^2$, $y^2$, and $xy+qyx$.
\begin{solution}
For $S=\kk$, $\theta_r(g\otimes f)(v\otimes w)=f(v)g(w)$. For $F=\alpha X\otimes X+\beta X\otimes Y+\gamma Y\otimes X+\delta Y\otimes Y$,
\[
  \theta_r(F)(x\otimes x)=\alpha,\qquad\theta_r(F)(y\otimes y)=\delta,\qquad\theta_r(F)(x\otimes y+q\,y\otimes x)=\gamma+q\beta.
\]
So $R^\perp=\kk(X\otimes Y-qY\otimes X)$ and $C_q^!=\kk\langle X,Y\rangle/(XY-qYX)$. Using $XY=qYX$, every word is a multiple of $Y^aX^b$, so these span; they are independent:
for $q=0$ the relation is monomial and $\{Y^aX^b\}$ is the set of words avoiding $XY$; for $q\neq0$ let $X$ act on $\kk[s,t]$ by multiplication by $s$ and $Y$ by $(Yf)(s,t)=t\,f(q^{-1}s,t)$; then $XY=qYX$ and $Y^aX^b\cdot1=q^{-ab}s^bt^a$, which are independent.
Hence $h_{C_q^!}=\sum_d(d+1)t^d=(1-t)^{-2}$ and $h_{C_q}(t)h_{C_q^!}(-t)=(1+t)^2(1+t)^{-2}=1$.
$C_1^!=\kk[X,Y]$ is commutative while in $C_0^!=\kk\langle X,Y\rangle/(XY)$ one has $XY=0\neq YX$: same Hilbert series, non-isomorphic algebras.
\end{solution}
\end{subparts}
\end{parts}
