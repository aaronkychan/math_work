\textbf{Exercises from Lectures 1--3.}
Throughout, $\kk$ is a field, and all modules are right modules unless a
left action is specified. Write $\kk A_n$ for the algebra of upper triangular
$n\times n$ matrices over $\kk$, with matrix units $e_{ij}$ ($1\leq i\leq j\leq n$)
and grading $\deg e_{ij}=j-i$.
For a graded module $M$, the shift convention is $M(r)_d=M_{d+r}$.
Ideals in algebra presentations are two-sided ideals.

\section*{Lecture 1: Graded algebras, modules, and duals}
\begin{questions}

\titledquestion{Graded modules and homogeneous maps}
The following parts are independent.
\begin{parts}
\part Let $n\geq1$ and $A=\kk A_n$.
Verify that
\[
  A_d=\operatorname{span}_{\kk}\{e_{ij}\mid j-i=d\}
\]
is a grading, using $e_{ij}e_{pq}=\delta_{jp}e_{iq}$.
Compute $\rad(A)$ from its definition as the intersection of all maximal
right ideals.

\part Let $A=\kk A_3$ and $M=e_{22}A$.
Its graded linear dual is
\[
  \bbD(M)_d=\Hom_{\kk}(M_{-d},\kk),\qquad
  (af)(m)=f(ma)
  \quad(a\in A,\ f\in\bbD(M),\ m\in M).
\]
Find a homogeneous basis of $\bbD(M)$, its degrees, and all non-zero
actions of the matrix units on that basis.
Show that $\bbD(M)$ is not isomorphic to any left module $(Ae_{jj})(r)$,
for $j\in\{1,2,3\}$ and $r\in\mathbb Z$.

\emph{Hint.} Compare dimensions and the actions of $e_{11},e_{22},e_{33}$.

\part Let $S=\kk e_1\oplus\kk e_2$, with $e_ie_j=\delta_{ij}e_i$ and
$1=e_1+e_2$, in degree $0$. Set
\[
  M=\bigoplus_{i\geq0}\kk m_i,\qquad
  \deg m_i=i,\quad m_ie_1=m_i,\quad m_ie_2=0,
\]
and let $N=\kk e_1$ be in degree $0$.
Define the $S$-linear map $f:M\to N$ by $f(m_i)=e_1$ for every $i\geq0$.
Write every non-zero homogeneous component $f_d:M\to N$ explicitly,
where a degree $d$ map sends $M_j$ into $N_{j+d}$.
Prove that $f$ is not a finite sum of homogeneous maps.

\part 
Let $A=\kk A_3$ and $P_i=e_{ii}A$ for $i\in\{1,2,3\}$, with the
inherited grading.
\begin{subparts}
\subpart For each of the nine ordered pairs $(i,j)$, find a basis of
$\Hom_A(P_i,P_j)$ and the degree of every basis map.
Present the answers in a $3\times3$ table whose rows index the source $P_i$
and whose columns index the target $P_j$.
\subpart Compute the composite
\[
  P_3\xrightarrow{\,p\mapsto e_{23}p\,}P_2
     \xrightarrow{\,p\mapsto e_{12}p\,}P_1.
\]
Determine the degree of each map and of the composite.
\subpart For every $i,j\in\{1,2,3\}$ and $r,s\in\mathbb Z$, determine
when there is a non-zero degree-preserving map $P_i(r)\to P_j(s)$.
\end{subparts}
\emph{Hint.} An $A$-linear map $u:P_i\to P_j$ is determined by
$u(e_{ii})$, and $u(e_{ii})=u(e_{ii})e_{ii}$.

\end{parts}

\titledquestion{Dependence on the symmetrising form}
Let $A=\kk A_2$, $S=\kk e_{11}\oplus\kk e_{22}$, $V=\kk e_{12}$,
and $M=e_{11}A$, with their multiplication actions and inherited gradings.
For $c_1,c_2\in\kk\setminus\{0\}$, define
\[
  \tau:S\longrightarrow\kk,\qquad
  ae_{11}+be_{22}\longmapsto c_1a+c_2b.
\]
Recall that $M^{\grstar}=\bigoplus_{d\in\mathbb Z}\Hom_S(M_{-d},S)$,
with the displayed summand in degree $d$.
A form is symmetrising when $\tau(st)=\tau(ts)$ for all $s,t\in S$
and the pairing $(s,t)\mapsto\tau(st)$ is non-degenerate.
For parts (3)--(4), take $\kk=\mathbb Q$.
\begin{parts}
\part Verify that $\tau$ is symmetrising. Define
$\varepsilon:V\to\kk$ by $\varepsilon(ue_{12})=u$ for $u\in\kk$.
Find the right $S$-linear map $f:V\to S$ and the left $S$-linear map
$g:V\to S$ satisfying $\tau\circ f=\varepsilon=\tau\circ g$.
Verify the stated linearities and determine whether $f$ is also left
$S$-linear and whether $g$ is also right $S$-linear.

\emph{Check.} $f(e_{12})=c_2^{-1}e_{22}$ and $g(e_{12})=c_1^{-1}e_{11}$.

\part Consider the right $S$-linear maps
\[
\begin{aligned}
  \varphi_0:M&\longrightarrow S,& ae_{11}+be_{12}&\longmapsto ae_{11},\\
  \varphi_1:M&\longrightarrow S,& ae_{11}+be_{12}&\longmapsto be_{22}
  \qquad(a,b\in\kk).
\end{aligned}
\]
Verify that $\varphi_0,\varphi_1$ form a homogeneous basis of
$M^{\grstar}$, and determine their degrees.

\part Set $\tau_1(ae_{11}+be_{22})=a+b$ and
$\tau_2(ae_{11}+be_{22})=a+2b$.
For $i\in\{1,2\}$, define
\[
  \alpha_i:M^{\grstar}\longrightarrow\bbD(M),\quad f\longmapsto\tau_i\circ f,
  \qquad a\mathbin{\cdot_i}f=\alpha_i^{-1}(a\alpha_i(f)),
\]
where $(a\lambda)(m)=\lambda(ma)$ for $a\in A$, $\lambda\in\bbD(M)$,
and $m\in M$.
Calculate the actions of $e_{11},e_{22},e_{12}$ on $\varphi_0,\varphi_1$
for both forms. Are the two actions equal?

\part Construct a graded left $A$-module isomorphism
\[
  t:(M^{\grstar},\cdot_1)\longrightarrow(M^{\grstar},\cdot_2)
\]
by specifying $t(\varphi_0)$ and $t(\varphi_1)$.
\end{parts}
\end{questions}

\titledquestion{Interaction of tensor product with $S$-dual}
Let $S$ be a finite-dimensional semisimple $\kk$-algebra, and let $V,W$
be finite-dimensional $S$-bimodules. For any such bimodule $U$, let
\[
  U^*=\Hom_S(U,S)
\]
be the space of right $S$-linear maps, with bimodule action
\[
  (sft)(u)=sf(tu)
  \qquad(s,t\in S,\ f\in U^*,\ u\in U).
\]
For $f\in V^*$, $g\in W^*$, $v\in V$, and $w\in W$, consider
\[
\begin{aligned}
  \theta_r^{V,W}:W^*\otimes_S V^*&\longrightarrow(V\otimes_S W)^*,\\
  \theta_r^{V,W}(g\otimes f)(v\otimes w)&=g(f(v)w).
\end{aligned}
\]
Prove that $\theta_r^{V,W}$ is a natural isomorphism of $S$-bimodules.
\begin{parts}
\part Prove that the formula defines an $S$-bimodule map.
Explicitly, check bilinearity in $(g,f)$ and in $(v,w)$, and show that
neither of the substitutions
\[
  vs\otimes w\ \longleftrightarrow\ v\otimes sw,
  \qquad
  gs\otimes f\ \longleftrightarrow\ g\otimes sf
  \qquad(s\in S)
\]
changes the value given by the formula.
Check that the resulting map $V\otimes_S W\to S$ is right $S$-linear,
and verify
\[
  \theta_r^{V,W}(sg\otimes ft)
  =s\,\theta_r^{V,W}(g\otimes f)t
  \qquad(s,t\in S).
\]
\part Prove that $\theta_r^{V,W}$ is bijective.
\part For finite-dimensional $S$-bimodules $V',W'$ and bimodule maps
$p:V\to V'$ and $q:W\to W'$, prove
\[
  (p\otimes q)^*\circ\theta_r^{V',W'}
     =\theta_r^{V,W}\circ(q^*\otimes p^*).
\]
Here $p^*:(V')^*\to V^*$ sends $f$ to $f\circ p$;
$q^*:(W')^*\to W^*$ and $(p\otimes q)^*:(V'\otimes_S W')^*\to(V\otimes_S W)^*$
are defined in the same way by composition with $q$ and $p\otimes q$,
respectively.
\end{parts}

\emph{Hint for (2).} First take $V=S$. Then use the fact that every
finite-dimensional right $S$-module is a direct summand of $S^{\oplus n}$
for some $n\geq0$.

\emph{Remark.} The analogous statement holds for left $S$-duals,
with left and right interchanged.

\titledquestion{Quadratic presentations and Hilbert series}
The following parts are independent. Throughout this sheet, for an algebra
$A=\bigoplus_{d\geq0}A_d$ with $A_0=\bigoplus_{i=1}^r\kk e_i$, use
\[
  H_A(t)_{ij}=\sum_{d\geq0}\dim_{\kk}(e_iA_de_j)t^d
  \qquad(1\leq i,j\leq r),
\]
with the idempotents in their listed order. Thus the coefficient of $t^d$
is the dimension vector of the $A_0$-bimodule $A_d$, recorded with two
indices as in Lecture~2. When $A_0=\kk$, there is just one entry.
\begin{parts}
\part Let $A=\kk A_4/(e_{13},e_{24})$ with the quotient grading.
Find a homogeneous basis, compute $H_A(t)$, and write
$A=T_S(V)/(R)$ with explicit $S$, $V$, and $R\subset V\otimes_S V$.

\part Let $S=\kk e_1\oplus\kk e_2$ with coordinate idempotents, and let
$V=\kk a\oplus\kk b$ satisfy
\[
  e_1ae_2=a,\qquad e_2be_1=b,
\]
with all other idempotent actions zero. Place $V$ in degree $1$.
Find a basis of $T_S(V)_d$ for every $d\geq0$, and show that
\[
  H_{T_S(V)}(t)=\frac{1}{1-t^2}\begin{pmatrix}1&t\\t&1\end{pmatrix}.
\]
Find a homogeneous basis and the Hilbert series of $T_S(V)/(ab)$,
and compare the two algebras.

\emph{Hint.} Count alternating words separately in each degree,
including degree $0$.

\part Let $V$ and $W$ have bases $x_1,\ldots,x_n$ and $y_1,\ldots,y_m$,
respectively. Give every generator degree $1$ and set
\[
\begin{aligned}
  \Sym(V)&=T_{\kk}(V)/(x_ix_j-x_jx_i\mid 1\leq i<j\leq n),\\
  \bigwedge W&=T_{\kk}(W)/(w^2\mid w\in W).
\end{aligned}
\]
Prove that
\[
  \dim_{\kk}\Sym(V)_2=\frac{n(n+1)}2,\qquad
  \dim_{\kk}(\bigwedge W)_2=\frac{m(m-1)}2.
\]
\end{parts}
\end{questions}

\section*{Lecture 3: The quadratic dual}
For $A=T_S(V)/(R)$, use the right quadratic dual
$A^!=T_S(V^*)/(R^\perp)$, with $V^*=\Hom_S(V,S)$ placed in degree $1$.
The reversed pairing is
\[
  \theta_r(g\otimes f)(v\otimes w)=g(f(v)w).
\]
When $S=\bigoplus_i\kk e_i$ and $v\in e_iVe_j$ is a chosen basis vector,
its dual $v^*:V\to S$ takes $v$ to $e_j$ and the other basis vectors
to zero; it satisfies $e_jv^*e_i=v^*$.

\begin{questions}
\setcounter{question}{4}

\titledquestion{Computing quadratic duals}
The following parts are independent.
\begin{parts}
\part \textbf{Reverse generators and complement relations.}
Let $A=\kk A_4/(e_{13})$.
Write a presentation of $A^!$ using reversed generators, and find
homogeneous bases of $A$ and $A^!$.
Check directly that dualising twice recovers $A$, identifying each twice
reversed generator with its original matrix unit.

\emph{Hint.} Among $e_{12}e_{23}$ and $e_{23}e_{34}$, only the first is
a relation. Complement the forbidden composable pairs and reverse
their order.

\part \textbf{All products of two generators vanish.}
Let $S=\kk e_1\oplus\kk e_2$ and $V=\kk a\oplus\kk b$, with
$e_1ae_2=a$ and $e_2be_1=b$ and all other idempotent actions zero.
Compute the quadratic dual of $T_S(V)/(ab,ba)$ and the Hilbert series
of both algebras.

\emph{Check.} In the order $e_1,e_2$, the two series are
\[
  \begin{pmatrix}1&t\\t&1\end{pmatrix}
  \qquad\text{and}\qquad
  \frac{1}{1-t^2}\begin{pmatrix}1&t\\t&1\end{pmatrix}.
\]

\part \textbf{When does every linear combination square to zero?}
Write $\kk\langle x,y\rangle:=T_{\kk}(\kk x\oplus\kk y)$; its basis
consists of the finite words in $x,y$, including the empty word $1$,
and multiplication joins words in order.
For $q\in\kk$, let
\[
  C_q=\kk\langle x,y\rangle/(x^2,y^2,xy+qyx),
  \qquad \deg x=\deg y=1.
\]
\begin{subparts}
\subpart Find a homogeneous basis of $C_q$, compute every product of two
basis elements, and determine $H_{C_q}(t)$. Treat $q=0$ as well as $q\neq0$, and justify
that your proposed basis is linearly independent.
\subpart For $u=\alpha x+\beta y\in(C_q)_1$, compute $u^2$.
Determine, for each $q$, exactly which pairs $(\alpha,\beta)\in\kk^2$
give $u^2=0$. For which $q$ does every element of $(C_q)_1$ square to zero?
\subpart Compute a quadratic presentation of $C_q^!$ using the reversed
pairing. Identify $C_1$ and $C_{-1}$ as the graded and ordinary tensor
products, respectively, of two copies of $\kk[z]/(z^2)$ with $\deg z=1$.
When the characteristic of $\kk$ is not $2$, are $C_1$ and $C_{-1}$
isomorphic as algebras? Explain what changes in characteristic $2$.
\end{subparts}
\emph{Hint.} In a candidate basis, use the class of $yx$ for degree $2$.
For the dual relation, evaluate a general linear combination of
$X^2,XY,YX,Y^2$ on $x^2,y^2,xy+qyx$, where $X=x^*$ and $Y=y^*$.
\end{parts}

\titledquestion{Bases, Hilbert series, and quadratic duals}
The following parts are independent.
\begin{parts}
\part \textbf{A square with a parameter.}
Let $S=\kk e_1\oplus\kk e_2\oplus\kk e_3\oplus\kk e_4$ and let
$V$ have basis $a,b,c,d$, with
\[
  e_1ae_2=a,\qquad e_2be_4=b,\qquad
  e_1ce_3=c,\qquad e_3de_4=d,
\]
and all other idempotent actions zero. The only composable words of
length $2$ are $ab$ and $cd$.
For each $\lambda\in\kk$, set
\[
  A_\lambda=T_S(V)/(ab-\lambda cd).
\]
For every $\lambda\in\kk$, including $\lambda=0$, compute a presentation
of $A_\lambda^!$. Find homogeneous bases of $A_\lambda$ and $A_\lambda^!$,
and compute both Hilbert series. Do the series depend on $\lambda$?
Recover the original relation by dualising again.
For $\lambda\neq0$, rescale $d$ to compare $A_\lambda$ with $A_1$.

\emph{Hint.} Evaluate $\mu b^*a^*+\nu d^*c^*$ on $ab-\lambda cd$,
where $\mu,\nu\in\kk$.

\part \textbf{Counting words.}
Let $V=\kk x\oplus\kk y$ and
\[
  B=T_{\kk}(V)/(x^2),\qquad \deg x=\deg y=1.
\]
\begin{subparts}
\subpart Describe a basis of $B_d$ for every $d\geq0$, and compute
$a_d:=\dim_{\kk}B_d$ for $0\leq d\leq6$.
\subpart For $d\geq2$, find an equation expressing $a_d$ in terms of
$a_{d-1}$ and $a_{d-2}$ by sorting the basis words according to their
first letter. Use the equation and the values of $a_0,a_1$ to find
$H_B(t)$ as a rational function.
\subpart Write $X=x^*$ and $Y=y^*$. Compute $B^!$, find a homogeneous
basis, and determine $H_{B^!}(t)$. Which of $B$ and $B^!$ are
finite-dimensional?
\end{subparts}
\end{parts}
\end{questions}
