$A^\en$ & Enveloping algebra $A\otimes_\Bbbk A^\op$, acting on the right by $m(a\otimes b^\op)=bma$. & \hyperref[def:enveloping-algebra]{\pageref*{def:enveloping-algebra}}\\[3pt]

\multicolumn{3}{@{}l}{\textbf{Presentations and Hilbert series}}\\*[4pt]
$\mu_i:A_1^{\otimes_S i}\to A_i$ & Multiplication of $i$ elements of degree 1. & \hyperref[eq:multiplication-maps]{\pageref*{eq:multiplication-maps}}\\[3pt]
$T_S(V)$ & Tensor algebra; degree zero part $S$ and degree $i$ part $V^{\otimes_S i}$ for $i\geq1$. & \hyperref[def:tensor-algebra]{\pageref*{def:tensor-algebra}}\\[3pt]
$\SymAlg(V),\ \bigwedge W$ & Symmetric algebra of $V$; exterior algebra of $W$. & \hyperref[ex:polynomial-exterior]{\pageref*{ex:polynomial-exterior}}\\[3pt]
$\mu:T_S(A_1)\to A$ & Canonical multiplication map for an algebra generated in degree 1. & \hyperref[cor:tensor-presentation]{\pageref*{cor:tensor-presentation}}\\[3pt]
$I=\Ker\mu$ & Homogeneous ideal of relations. & \hyperref[def:relations]{\pageref*{def:relations}}\\[3pt]
$A=T_S(V)/(R)$ & Quadratic presentation: $V$ consists of degree 1 generators, and $R\subset V\otimes_S V$ of degree 2 relations. & \hyperref[def:quadratic]{\pageref*{def:quadratic}}\\[3pt]
$(R)$ & Two-sided ideal generated by the relation space $R$. & \hyperref[def:quadratic]{\pageref*{def:quadratic}}\\[3pt]
$L_j=e_jS$ & Simple right $S$-module, viewed as $e_jA/e_jA_+$ in degree zero. & \hyperref[not:hilbert-simples-summands]{\pageref*{not:hilbert-simples-summands}}\\[3pt]
$[M_d:L_j]$ & Multiplicity of the composition factor $L_j$ in degree $d$. & \hyperref[def:dimension-vector]{\pageref*{def:dimension-vector}}\\[3pt]
$\dimvec M_d$ & Row vector $([M_d:L_j])_{1\leq j\leq n}$. & \hyperref[def:dimension-vector]{\pageref*{def:dimension-vector}}\\[3pt]
$h_M(t)$ & Module Hilbert series $\sum_{d\in\Z}(\dimvec M_d)t^d$. & \hyperref[def:hilbert-series]{\pageref*{def:hilbert-series}}\\[3pt]
$h_A(t)$ & Algebra Hilbert series; entry $(i,j)$ is $\sum_{d\geq0}[e_iA_d:L_j]t^d$. & \hyperref[def:hilbert-series]{\pageref*{def:hilbert-series}}\\[3pt]

\multicolumn{3}{@{}l}{\textbf{Quadratic duality and algebra tensor products}}\\*[4pt]
$\theta$ & Factor-reversing comparison $D(W)\otimes_S D(V)\to D(V\otimes_S W)$ for finite-dimensional bimodules over the standing $S$. & \hyperref[eq:D-tensor-duals]{\pageref*{eq:D-tensor-duals}}\\[3pt]
$U^\perp$ & Annihilator of $U\subset M$ in $D(M)$. & \hyperref[def:D-orthogonal]{\pageref*{def:D-orthogonal}}\\[3pt]
$R^\perp$ & Relation annihilator, viewed in $D(V)\otimes_S D(V)$ using $\theta$. & \hyperref[def:quadratic-duals]{\pageref*{def:quadratic-duals}}\\[3pt]
$A^!$ & Quadratic dual $T_S(D(V))/(R^\perp)$, with $D(V)$ placed in degree 1. & \hyperref[def:quadratic-duals]{\pageref*{def:quadratic-duals}}\\[3pt]
$A\otimes_\Bbbk B$ & Ordinary algebra tensor product, with multiplication $(a\otimes b)(a'\otimes b')=aa'\otimes bb'$. & \hyperref[lem:hilbert-field-tensor]{\pageref*{lem:hilbert-field-tensor}}\\[3pt]

\multicolumn{3}{@{}l}{\textbf{Tops, resolutions, and the Koszul complex}}\\*[4pt]
$\top M$ & Ordinary top $M/\rad M$, for finite-dimensional $A$. & \hyperref[def:module-top]{\pageref*{def:module-top}}\\[3pt]
$p:P\to M$ & Projective cover: no proper submodule of $P$ maps onto $M$. & \hyperref[def:projective-cover]{\pageref*{def:projective-cover}}\\[3pt]
$C_\bullet,\ d_i$ & Chain complex with differentials $d_i:C_i\to C_{i-1}$ and $d_id_{i+1}=0$. & \hyperref[def:chain-complex]{\pageref*{def:chain-complex}}\\[3pt]
$H_i(C_\bullet)$ & Homology $\Ker d_i/\Im d_{i+1}$. & \hyperref[def:chain-complex]{\pageref*{def:chain-complex}}\\[3pt]
$P_\bullet\to M$ & Projective resolution; $P_\bullet$ ends in $P_0\to0$, with a specified map $p_0:P_0\to M$. & \hyperref[def:projective-resolution]{\pageref*{def:projective-resolution}}\\[3pt]
$\grtop M$ & Graded top $M/MA_+$. & \hyperref[not:graded-top]{\pageref*{not:graded-top}}\\[3pt]
$K_i$ & Koszul term $\bigwedge^iV\otimes_\Bbbk A(-i)$ for $A=\SymAlg(V)$; the exterior factor is placed in degree zero. & \hyperref[not:koszul-terms]{\pageref*{not:koszul-terms}}\\[3pt]
$K_\bullet$ & Koszul complex, with degree-preserving differentials $d_i:K_i\to K_{i-1}$. & \hyperref[def:koszul-complex]{\pageref*{def:koszul-complex}}\\[3pt]

\multicolumn{3}{@{}l}{\textbf{Advanced notation}}\\*[4pt]
$\add_{\mod A^\op}(M)$ & Additive closure in the category of left $A$-modules. & \hyperref[prop:tensor-duals]{\pageref*{prop:tensor-duals}}\\[3pt]
$M^*,\ {}^*N$ & Right and left $S$-duals: $\Hom_S(M,S)$ and $\Hom_{S^\op}(N,S)$. & \hyperref[def:S-duals]{\pageref*{def:S-duals}}\\[3pt]
$M^{\grstar},\ {}^{\grstar}N$ & Graded right and left $S$-duals; degree $i$ is the corresponding dual of degree $-i$. & \hyperref[def:graded-S-duals]{\pageref*{def:graded-S-duals}}\\[3pt]
