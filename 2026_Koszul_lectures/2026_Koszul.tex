\section*{Convention}

Throughout the course, $\Bbbk$ is a field.
All rings are unital and associative.
All modules are right modules unless otherwise specified.
When we discuss dimensions, we mean dimensions as $\Bbbk$-vector spaces.


\section{Rings and modules}

\subsection{Semisimple algebras}

\begin{definition}
A \defn{$\Bbbk$-algebra} is a $\Bbbk$-vector space with an associative unital multiplication that is \defn{bilinear}, meaning linear in each variable.
\end{definition}

A right $A$-module is understood to be a $\Bbbk$-vector space $M$ with a bilinear action satisfying $(ma)b=m(ab)$ and $m1_A=m$ for $m\in M$ and $a,b\in A$.

\begin{example}\label{ex:triangular-algebra}
(1) The matrix ring $\Mat_n(\Bbbk)$ is a $\Bbbk$-algebra.
Every row space (the space of $1\times n$-matrices) is a $\Mat_n(\Bbbk)$-module.

(2) Let $\Bbbk A_n$ be the \defn{upper triangular matrix ring}, which is also a $\Bbbk$-algebra.
\[
\Bbbk A_n = \begin{pmatrix}
\Bbbk & \Bbbk & \cdots & \Bbbk \\
0 & \Bbbk & \cdots & \vdots \\
0 & 0 & \ddots & \vdots \\
0 & \cdots & 0 & \Bbbk
\end{pmatrix}
\]
For those familiar with path algebras, $\Bbbk A_n$ is the path algebra of a \defn{linearly oriented $A_n$ quiver}, which explains the notation. We use matrix calculations, so no knowledge of quivers is required.
\end{example}

\begin{definition}
A module $L$ is \defn{simple} if $L\neq 0$ and its only submodules are $0$ and $L$.
A module is \defn{semisimple} if it is a direct sum of simple modules.
An algebra $B$ is \defn{semisimple} if it is semisimple as a module $B_B$ over itself (this does not depend on which side you work on).
The \defn{Jacobson radical} $\rad B$ is the intersection of all maximal right ideals of $B$.
\end{definition}

We use the following standard facts without proof.
Recall that a \defn{division $\Bbbk$-algebra} is a non-zero $\Bbbk$-algebra whose non-zero elements are invertible.

\begin{theorem}[Artin--Wedderburn]\label{thm:artin-wedderburn}\label{prop:semisimple}
For a non-zero finite-dimensional $\Bbbk$-algebra $B$,
\[
B\text{ is semisimple}
\quad\Longleftrightarrow\quad
\rad B=0
\quad\Longleftrightarrow\quad
B\cong\prod_{i=1}^r\Mat_{n_i}(D_i),
\]
where $r,n_i\geq 1$ and each $D_i$ is a finite-dimensional division $\Bbbk$-algebra.
If $\Bbbk$ is algebraically closed, then $D_i=\Bbbk$ for every $i$.
For semisimple $B$, every finite-dimensional left or right $B$-module is semisimple, and every submodule has a complement.
\end{theorem}

%When all $D_i=\Bbbk$, the algebra is called \defn{split semisimple}: it is a product of independent matrix blocks.
%For example, $\Mat_2(\Bbbk)\times\Mat_3(\Bbbk)$ is the algebra of block-diagonal matrices with blocks of sizes $2$ and $3$.
%We do not require semisimple algebras to be split.

\begin{example}\label{ex:semisimple-algebras}
Let $e_{ij}$ be the unit matrix with $1$ at the $(i,j)$-entry and zero everywhere else.   Then we have $\rad(\Bbbk A_2)=\Bbbk e_{12}\neq 0$, and so $\Bbbk A_2$ is not semisimple.
\end{example}

\subsection{Graded algebras}

\begin{definition}\label{def:group-grading}
Let $G$ be a group with identity $e$.
A \defn{$G$-graded $\Bbbk$-algebra} is a $\Bbbk$-algebra $A$ with $1_A\in A_e$ and a decomposition
\[
A=\bigoplus_{g\in G}A_g
\quad\text{such that}\quad
A_gA_h\subset A_{gh}
\quad\text{for all }g,h\in G.
\]
An element of $A_g$ is \defn{homogeneous of degree $g$}.
\end{definition}

\begin{convention}
Unless stated otherwise, a graded algebra is $\Z$-graded and \uline{concentrated in non-negative degrees}.
\end{convention}

Note that the decomposition is into a direct sum, so every element of a graded algebra $A$ can always be written as a \emph{finite} sum of homogeneous elements.

\begin{example}\label{ex:triangular-grading}
For the upper triangular matrix ring $\Bbbk A_n$ of Example~\ref{ex:triangular-algebra}, the decomposition
\[
(\Bbbk A_n)_d=\operatorname{span}_\Bbbk\{e_{ij}\mid j-i=d\}
\]
makes $\Bbbk A_n$ a graded algebra.
\end{example}


\begin{example}
The algebra $\Bbbk A_3$ also has a $\Z^{\oplus 2}$-grading with
\[
\deg e_{ii}=(0,0),\qquad
\deg e_{12}=(1,0),\qquad
\deg e_{23}=(0,1),\qquad
\deg e_{13}=(1,1).
\]
The homomorphism $\Z^{\oplus 2}\to\Z$, $(p,q)\mapsto p+q$, gives the grading of Example~\ref{ex:triangular-grading}.
Alternatively, setting $\deg e_{ij}=2(j-i)$ gives a $\Z$-grading with $A_1=0$ but $A_2\neq 0$.
\end{example}

\begin{definition}
A \defn{homomorphism of graded algebras} $f:A\to B$ is a $\Bbbk$-algebra homomorphism with $f(A_i)\subset B_i$ for all $i\in\Z$.
\end{definition}

\begin{definition}
A graded algebra $A$ is \defn{locally finite} if $\dim_\Bbbk A_i<\infty$ for every $i\in\Z$.
\end{definition}

\subsection{Graded modules}

\begin{definition}
Let $A=\bigoplus_{g\in G}A_g$ be a $G$-graded algebra.
A \defn{$G$-graded right $A$-module} is a right $A$-module $M$ with a direct sum decomposition
\[
M=\bigoplus_{g\in G}M_g
\quad\text{such that}\quad
M_gA_h\subset M_{gh}
\quad\text{for all }g,h\in G.
\]
An element of $M_g$ is called \defn{homogeneous of degree $g$}.

A $G$-graded $A$-module $M=\bigoplus_{g\in G} M_g$ is \defn{locally finite-dimensional} if $\dim_{\Bbbk} M_g <\infty$ for all $g\in G$.
\end{definition}

Unless specified otherwise, graded modules are $\Z$-graded; their degrees need not be non-negative.
Write $\Mod A$ for the category of right $A$-modules, $\Grmod A$ for graded right $A$-modules with degree $0$ homomorphisms, and $\grmod A$ for its full subcategory of locally finite-dimensional modules.

The \defn{opposite algebra} $A^\op$ has the same vector space as $A$, with multiplication $a^\op b^\op=(ba)^\op$.
A left $A$-module is a right $A^\op$-module by $m\cdot a^\op=am$.
For a $\Z$-graded algebra, $(A^\op)_i=A_i$.

\begin{definition}\label{def:bimodule}
An \defn{$A$-bimodule} is a vector space $M$ with $\Bbbk$-bilinear left and right $A$-module actions satisfying $(am)b=a(mb)$ for all $a,b\in A$ and $m\in M$.
It is graded if $M=\bigoplus_{j\in\Z}M_j$ and $A_iM_jA_k\subset M_{i+j+k}$ for all $i,j,k\in\Z$.
\end{definition}

The algebra $A$ is a graded $A$-bimodule by multiplication, and each $A_i$ is an $A_0$-bimodule.
The left and right scalar actions agree: $(c1_A)m=cm=m(c1_A)$ for $c\in\Bbbk$.

\begin{example}
The decomposition $A=\bigoplus_{i\in\Z}A_i$ makes the right regular module $A_A$ a graded module.
\end{example}

\begin{definition}\label{def:shift}
Let $M$ be a graded $A$-module and let $d\in\Z$.
The \defn{shift} $M(d)$ is the graded $A$-module whose underlying $A$-module is $M$, but whose grading is
\[
M(d)_j=M_{j+d}.
\]
\end{definition}

By Definition~\ref{def:shift}, $1\in A_0=A(-d)_d$, so $A(-d)$ has a generator in degree $d$.

\begin{example}\label{ex:shifted-row}
For $A=\Bbbk A_3$ with the grading of Example~\ref{ex:triangular-grading}, the module $P_1:=e_{11}A$ consists of matrices supported on the first row.
Its basis $e_{11},e_{12},e_{13}$ has degrees $0,1,2$, respectively; in $P_1(-2)$, the same basis has degrees $2,3,4$.
\end{example}

\begin{definition}
Let $M,N$ be graded $A$-modules.
A homomorphism $f:M\to N$ is \defn{homogeneous of degree $d$} if
$f(M_j)\subset N_{j+d}$ for all $j\in\Z$. A \defn{homomorphism of graded modules} means a homogeneous $A$-module homomorphism of degree $0$, also said to \defn{preserve degrees}.
\end{definition}

\begin{notation}\label{not:graded-hom}
For graded right $A$-modules $M,N$, $\Hom_A(M,N)$ denotes all right $A$-module homomorphisms, without a grading condition. We use
\begin{equation}\label{eq:graded-hom}
\begin{aligned}
\Hom_{\Grmod A}(M,N)&=\{f\in\Hom_A(M,N)\mid f(M_j)\subset N_j\text{ for all }j\in\Z\},\\
\underline{\Hom}_A(M,N)&=\bigoplus_{d\in\Z}\Hom_{\Grmod A}(M,N(d)).
\end{aligned}
\end{equation}
The degree $d$ component of $\underline{\Hom}_A(M,N)$ consists of homomorphisms $M\to N$ of degree $d$, since $N(d)_j=N_{j+d}$.
For left modules we use $\Hom_{A^\op}$ and $\Hom_{\Grmod A^\op}$.
\end{notation}

There is a natural inclusion $\underline{\Hom}_A(M,N)\subset\Hom_A(M,N)$: an element on the left is a finite sum of homogeneous homomorphisms. Equality holds when $M$ is finitely generated, but need not hold for arbitrary $M$.

\begin{example}\label{ex:matrix-hom}
For $A=\Bbbk A_3$, left multiplication by $e_{12}$ gives
\[
\begin{aligned}
e_{22}A&\longrightarrow e_{11}A,\\[-2pt]
\begin{pmatrix}0&0&0\\0&u&v\\0&0&0\end{pmatrix}
&\longmapsto
\begin{pmatrix}0&u&v\\0&0&0\\0&0&0\end{pmatrix}
\qquad(u,v\in\Bbbk).
\end{aligned}
\]
The $u$ and $v$ entries have degrees $0,1$ in the source and $1,2$ in the target. Thus the map has degree $1$; after replacing the source by $e_{22}A(-1)$, it belongs to $\Hom_{\Grmod A}(e_{22}A(-1),e_{11}A)$.
Right $A$-linearity follows from associativity of matrix multiplication.
\end{example}

\subsection{Homogeneous ideals and quotients}

\begin{definition}
Let $A=\bigoplus_{i\in\Z}A_i$ be a graded algebra.
A right ideal $I\subset A$ is \defn{homogeneous} if
\[
I=\bigoplus_{i\in\Z}(I\cap A_i).
\]
Equivalently, whenever $a\in I$ and $a=a_0+\cdots+a_N$ is its decomposition into homogeneous components, each $a_i$ also belongs to $I$.
\end{definition}

For quotient algebras we need two-sided ideals.
If $I$ is a homogeneous two-sided ideal, then
\[
A/I=\bigoplus_{i\in\Z} A_i/(I\cap A_i)
\]
is again a graded algebra.

\begin{example}\label{ex:quadratic-matrix-quotient}
In $A=\Bbbk A_3$ with the grading of Example~\ref{ex:triangular-grading}, the two-sided ideal $I=(e_{13})=\Bbbk e_{13}$ is homogeneous.
Writing $\overline e_{ij}$ for the image of $e_{ij}$,
\[
(A/I)_0\cong\Bbbk^{\oplus 3},\qquad
(A/I)_1=\Bbbk\overline e_{12}\oplus\Bbbk\overline e_{23},\qquad
(A/I)_d=0\quad(d\geq 2).
\]
In particular, $\overline e_{12}\overline e_{23}=0$.
\end{example}

\begin{example}
In $A=\Bbbk A_2$, the subspace $I=\Bbbk(e_{22}+e_{12})$ is a right ideal: $Ie_{11}=Ie_{12}=0$ and $Ie_{22}=I$.
However, $e_{22},e_{12}\notin I$, so $I$ is not homogeneous.
The right module $A/I$ therefore does not inherit the quotient grading.
\end{example}

\begin{definition}
Let $A$ be a graded algebra and set
\[
A_{>0}:=\bigoplus_{i>0}A_i.
\]
The subspace $A_{>0}$ is a homogeneous two-sided ideal, also called the \defn{augmentation ideal}. We use the term \defn{positive degree ideal}.
\end{definition}

The quotient map, also called the augmentation map,
\[
\pi:A\longrightarrow A/A_{>0}\cong A_0,\qquad
\sum_{i\in\Z}a_i\mapsto a_0,
\]
makes $A_0$ a graded $A$-bimodule concentrated in degree $0$, with $a\cdot s\cdot b=\pi(a)s\pi(b)$ for $a,b\in A$ and $s\in A_0$.


\subsection{Tensor products of vector spaces}

\begin{definition}\label{def:vector-tensor}
For vector spaces $U,V$, the \defn{tensor product} $U\otimes_\Bbbk V$ is equipped with a bilinear map $(u,v)\mapsto u\otimes v$ such that every bilinear map $b:U\times V\to W$ has a unique linear factorisation:
\[
\xymatrix@C=60pt@R=22pt{
U\times V \ar[r]^{(u,v)\mapsto u\otimes v} \ar[dr]_{b}
& U\otimes_\Bbbk V \ar@{-->}[d]^{\exists!\,\widetilde b}\\
& W.
}
\]
\end{definition}

If $(u_i)_{i=1}^r$ and $(v_j)_{j=1}^s$ are bases, then $(u_i\otimes v_j)_{i,j}$ is a basis of $U\otimes_\Bbbk V$; hence $\dim_\Bbbk(U\otimes_\Bbbk V)=rs$.
We use the canonical identifications $(U\otimes_\Bbbk V)\otimes_\Bbbk W\cong U\otimes_\Bbbk(V\otimes_\Bbbk W)$ and $\Bbbk\otimes_\Bbbk U\cong U$, and write $U^{\otimes 0}=\Bbbk$.

\subsection{\texorpdfstring{Graded $\Bbbk$-linear duals}{Graded k-linear duals}}

A linear functional on a graded vector space need not have a degree: it may be non-zero on infinitely many homogeneous components.
To obtain a graded vector space, we retain only finite sums of homogeneous functionals.

\begin{definition}\label{def:linear-duals}
For a vector space $V$, its \defn{$\Bbbk$-linear dual} is $D(V):=\Hom_\Bbbk(V,\Bbbk)$.
For a graded vector space $V=\bigoplus_{i\in\Z}V_i$, its \defn{graded $\Bbbk$-linear dual} is the graded vector space given by
\[
\bbD(V)=\bigoplus_{i\in \Z} \bbD(V)_i
\qquad\text{with}\qquad
\bbD(V)_i=D(V_{-i}).
\]
\end{definition}

For general infinite-dimensional $V$, we have $\bbD(V)\subset D(V)$, whereas $D(V)\cong\prod_{i\in\Z}D(V_i)$.
These spaces coincide when $V$ is finite-dimensional.

Linear duality reverses the side of a module action.
For a right $A$-module $M$ and a left $A$-module $N$, define
\begin{equation}\label{eq:linear-dual-actions}
(a f)(m):=f(ma),\qquad (g a)(n):=g(an)
\quad(a\in A,\ f\in D(M),\ g\in D(N)).
\end{equation}
Here $m\in M$ and $n\in N$; the first formula makes $D(M)$ a left module and the second makes $D(N)$ a right module.
For graded modules, the same formulas preserve the graded $\Bbbk$-linear dual: if $a\in A_p$ and $f\in\bbD(M)_q$, then $af$ vanishes outside $M_{-p-q}$ and hence has degree $p+q$.
There are no signs in these actions.

\begin{example}\label{ex:row-linear-dual}
Let $A=\Bbbk A_3$ and $P_1=e_{11}A$ as in Example~\ref{ex:shifted-row}.
The functionals $\varepsilon_{1j}(e_{1\ell})=\delta_{j\ell}$ for $1\leq j,\ell\leq 3$ form a basis of $\bbD(P_1)$, with
\[
\deg\varepsilon_{1j}=1-j,\qquad
e_{12}\varepsilon_{12}=\varepsilon_{11},\qquad
e_{23}\varepsilon_{13}=\varepsilon_{12},\qquad
e_{jj}\varepsilon_{1j}=\varepsilon_{1j}.
\]
The actions follow from \eqref{eq:linear-dual-actions}.
For comparison with an infinite-dimensional space, take $V=\bigoplus_{i\geq 0}\Bbbk v_i$ with $\deg v_i=i$.
The functional $\lambda(v_i)=1$ for every $i$ belongs to $D(V)$ but not to $\bbD(V)$.
\end{example}

\begin{lemma}\label{lem:linear-bidual}
$\bbD$ is a duality between $\grmod A$ and $\grmod A^\op$: it reverses arrows and short exact sequences, and evaluation gives a graded module isomorphism $M\cong\bbD(\bbD(M))$ for every locally finite graded module $M$.
\end{lemma}

\begin{proof}
A homomorphism $u:M\to N$ of degree $0$ induces $\bbD(u):\bbD(N)\to\bbD(M)$ by $f\mapsto f\circ u$.
Dualising a short exact sequence in each degree is exact, since every linear functional on a subspace extends to the whole vector space.
The evaluation map sends $m$ to $(f\mapsto f(m))$.
It is an isomorphism in each degree by finite-dimensional linear algebra, and \eqref{eq:linear-dual-actions} makes it a module homomorphism.
\end{proof}

\subsection*{Left, right, and two-sided duals over $S$}

Fix a finite-dimensional semisimple $\Bbbk$-algebra $S$, concentrated in degree $0$.
Besides the $\Bbbk$-linear dual, we need duals defined by left $S$-linear maps, right $S$-linear maps, and maps respecting both actions.
Their definitions differ, even though the resulting spaces can be identified after a choice of form.

\begin{definition}\label{def:S-duals}
For a right $S$-module $M$ and a left $S$-module $N$, define their \defn{$S$-duals} by
\[
\begin{aligned}
M^*&:=\Hom_S(M,S),&
(sf)(m)&:=sf(m),\\
{}^*N&:=\Hom_{S^\op}(N,S),&
(gs)(n)&:=g(n)s.
\end{aligned}
\]
Here $s\in S$, $m\in M$, $n\in N$, $f\in M^*$, and $g\in{}^*N$.
\end{definition}

Thus $M^*$ is a left $S$-module and $ {}^*N$ is a right $S$-module.
The position of the star records the side of the original action: $f(ms)=f(m)s$ in $M^*$, whereas $g(sn)=sg(n)$ in $ {}^*N$.

\begin{definition}\label{def:enveloping-algebra}
The \defn{enveloping algebra} of $S$ is $S^{\mathrm{en}}:=S\otimes_\Bbbk S^\op$, with multiplication
\[
(s\otimes t^\op)(s'\otimes (t')^\op)=ss'\otimes(t't)^\op
\quad(s,t,s',t'\in S).
\]
\end{definition}

An $S$-bimodule is equivalently a right $S^{\mathrm{en}}$-module via
\begin{equation}\label{eq:enveloping-action}
m\cdot(s\otimes t^\op)=tms
\quad(s,t\in S,\ m\in M).
\end{equation}
For example, $(m\cdot(s\otimes t^\op))\cdot(s'\otimes(t')^\op)=t'tmss'=m\cdot(ss'\otimes(t't)^\op)$.
All three spaces $D(M)$, $M^*$, and $ {}^*M$ inherit bimodule structures:
\begin{equation}\label{eq:bimodule-dual-actions}
(s\lambda t)(m)=\lambda(tms),\qquad
(sft)(m)=sf(tm),\qquad
(sgt)(m)=g(ms)t,
\end{equation}
where $\lambda\in D(M)$, $f\in M^*$, $g\in{}^*M$, $s,t\in S$, and $m\in M$.

\begin{definition}\label{def:en-dual}
The \defn{two-sided dual} of an $S$-bimodule $M$ is
\[
M^{*_{\mathrm{en}}}:=\Hom_{S^{\mathrm{en}}}(M,S^{\mathrm{en}}),
\qquad (sFt)(m):=(s\otimes t^\op)F(m)
\quad(s,t\in S,\ F\in M^{*_{\mathrm{en}}},\ m\in M).
\]
\end{definition}

In Definition~\ref{def:en-dual}, $M$ has the right action \eqref{eq:enveloping-action}, so $F(tms)=F(m)(s\otimes t^\op)$.
The output is a left $S^{\mathrm{en}}$-module by multiplication on the values, which gives the displayed bimodule action.
The two-sided dual takes values in $S^{\mathrm{en}}$, not in $S$.

\subsection{Comparing the duals}

An $S$-dual consists of $S$-valued maps; the $\Bbbk$-linear dual consists of scalar-valued maps. A suitable linear functional $S\to\Bbbk$ identifies these constructions.

\begin{definition}\label{def:symmetrising-form}
A \defn{symmetrising form} on a finite-dimensional $\Bbbk$-algebra $S$ is a linear map $\tau:S\to\Bbbk$ satisfying
\[
\tau(st)=\tau(ts)\quad\text{for all }s,t\in S,
\qquad
\bigl(\tau(st)=0\text{ for all }t\in S\bigr)\ \Longrightarrow\ s=0.
\]
An algebra admitting such a form is \defn{symmetric}.
\end{definition}

We use the standard theorem that every finite-dimensional semisimple $\Bbbk$-algebra is symmetric.
We do not prove this structure theorem here.
The forms needed in our basic examples are explicit.

\begin{example}\label{ex:symmetrising-form}
For $S=\Bbbk^{\oplus r}$, take $\tau(a_1,\ldots,a_r)=\sum_i a_i$.
For $S=\Mat_n(\Bbbk)$, take $\tau(a)=\Tr(a)$.
Indeed, $\Tr(ab)=\Tr(ba)$ and $\Tr(ae_{ji})=a_{ij}$, so the pairing is non-degenerate in every characteristic.
More generally, the sum of the matrix traces works on every product $\prod_{j=1}^r\Mat_{n_j}(\Bbbk)$ in Theorem~\ref{thm:artin-wedderburn}.
\end{example}

\begin{proposition}\label{prop:dual-comparison}
Choose a symmetrising form $\tau$ on $S$.
For every finite-dimensional $S$-bimodule $M$, there are isomorphisms of $S$-bimodules
\begin{equation}\label{eq:dual-comparison}
\alpha_M:M^*\longrightarrow D(M),\quad f\longmapsto\tau\circ f,
\qquad
\beta_M:{}^*M\longrightarrow D(M),\quad g\longmapsto\tau\circ g.
\end{equation}
There is also an isomorphism $\gamma_M:M^{*_{\mathrm{en}}}\to M^*$ given by
\begin{equation}\label{eq:en-dual-comparison}
\gamma_M(F)(m)=\sum_j a_j\tau(b_j)
\quad\text{where}\quad
F(m)=\sum_j a_j\otimes b_j^\op.
\end{equation}
The maps $\alpha_M,\beta_M,\gamma_M$ commute with precomposition by bimodule homomorphisms.
The maps $\alpha_M$ and $\beta_M$ also apply to right and left $S$-modules, respectively, with the opposite side on the duals.
\end{proposition}

\begin{proof}
For $\lambda\in D(M)$, non-degeneracy of $\tau$ determines $f(m),g(m)\in S$ uniquely by
\begin{equation}\label{eq:dual-inverses}
\tau(f(m)s)=\lambda(ms),\qquad
\tau(sg(m))=\lambda(sm)
\quad(s\in S,\ m\in M).
\end{equation}
Equation~\eqref{eq:dual-inverses} gives $f(ms)=f(m)s$, $g(sm)=sg(m)$, and $\alpha_M(f)=\beta_M(g)=\lambda$.
Hence $\alpha_M,\beta_M$ are bijective.
For bimodule compatibility, use \eqref{eq:bimodule-dual-actions} and
\[
\tau(sf(tm))=\tau(f(tm)s)=\tau(f(tms)),\qquad
\tau(g(ms)t)=\tau(tg(ms))=\tau(g(tms)).
\]

The form $\tau_{\mathrm{en}}(a\otimes b^\op)=\tau(a)\tau(b)$ is symmetrising on $S^{\mathrm{en}}$.
The formula \eqref{eq:en-dual-comparison} satisfies
\[
\gamma_M(F)(ms)=\gamma_M(F)(m)s,\qquad
\alpha_M(\gamma_M(F))=\tau_{\mathrm{en}}\circ F.
\]
The argument \eqref{eq:dual-inverses}, applied to $S^{\mathrm{en}}$ and $\tau_{\mathrm{en}}$, makes $F\mapsto\tau_{\mathrm{en}}\circ F$ an isomorphism.
Thus $\gamma_M$ is an isomorphism.
Composition with $\tau$ and $\tau_{\mathrm{en}}$ commutes with precomposition, giving naturality.
\end{proof}

The identifications \eqref{eq:dual-comparison} and \eqref{eq:en-dual-comparison} depend on $\tau$.
For instance, $\tau(a_1,\ldots,a_r)=\sum_i c_i a_i$ is symmetrising on $\Bbbk^{\oplus r}$ for any $c_i\neq 0$.
No semisimplicity assumption on $S^{\mathrm{en}}$ was used.

\begin{definition}\label{def:graded-S-duals}
For a graded $S$-bimodule $M$, define its \defn{graded $S$-duals} and \defn{graded two-sided dual} by
\[
M^{(*)}:=\bigoplus_{i\in\Z}(M_{-i})^*,\qquad
{}^{(*)}M:=\bigoplus_{i\in\Z}{}^*(M_{-i}),\qquad
M^{(*_{\mathrm{en}})}:=\bigoplus_{i\in\Z}(M_{-i})^{*_{\mathrm{en}}},
\]
where each indicated summand has degree $i$.
The first construction also applies to graded right $S$-modules, and the second to graded left $S$-modules.
\end{definition}

Parentheses on the stars distinguish the graded constructions from the full $S$-duals. In the notation \eqref{eq:graded-hom},
\[
M^{(*)}=\underline{\Hom}_S(M,S),\qquad
{}^{(*)}M=\underline{\Hom}_{S^\op}(M,S).
\]

\begin{corollary}\label{cor:graded-dual-comparison}
For a locally finite graded $S$-bimodule $M$, a choice of symmetrising form gives natural isomorphisms of graded $S$-bimodules
\[
M^{(*)}\cong\bbD(M)\cong{}^{(*)}M\cong M^{(*_{\mathrm{en}})}.
\]
In particular, $\bbD(M)\cong M^{(*)}$ for a graded right $S$-module, and $\bbD(M)\cong{}^{(*)}M$ for a graded left $S$-module.
\end{corollary}

\begin{proof}
Apply Proposition~\ref{prop:dual-comparison} to $M_{-i}$ and take the direct sum over $i\in\Z$.
Every comparison preserves degree because $\tau$ has degree $0$.
\end{proof}

For a graded algebra $A$ with $S=A_0$, Corollary~\ref{cor:graded-dual-comparison} transports the $A$-action \eqref{eq:linear-dual-actions} to the graded $S$-duals of a locally finite graded $A$-module:
\[
af=\alpha_M^{-1}(a\alpha_M(f))\quad\text{for right }M,
\qquad
ga=\beta_M^{-1}(\beta_M(g)a)\quad\text{for left }M.
\]
Here $\alpha_M(f)=\tau\circ f$ and $\beta_M(g)=\tau\circ g$ are the restrictions of \eqref{eq:dual-comparison}; $a\in A$, $f\in M^{(*)}$, and $g\in{}^{(*)}M$.
The restrictions to $S$ agree with Definition~\ref{def:S-duals}.
These remain duals over $S$, not duals taking values in $A$.

\begin{example}\label{ex:dual-blocks}
Let $S=\Bbbk^{\oplus r}$ with coordinate idempotents $e_1,\ldots,e_r$ and the form $\tau(e_i)=1$.
For an $S$-bimodule $M$, the decompositions and the reversal of sides are
\[
M=\bigoplus_{i,j}e_iMe_j,
\qquad
e_jD(M)e_i\cong D(e_iMe_j)\cong e_jM^*e_i\cong e_j({}^*M)e_i.
\]
For the degree $1$ piece $V=\Bbbk e_{12}$ of $\Bbbk A_2$, we have $e_1Ve_2=V$.
Its graded $\Bbbk$-linear dual is spanned by $\varepsilon$ of degree $-1$, with $\varepsilon(e_{12})=1$ and $e_2\varepsilon e_1=\varepsilon$.
The corresponding elements of $V^*$ and $ {}^*V$ send $e_{12}$ to $e_2$ and $e_1$, respectively.
Thus equal scalar-valued functionals can correspond to different $S$-valued maps.
\end{example}

\begin{lemma}\label{lem:S-bidual}
For finite-dimensional right $S$-modules $M$ and left $S$-modules $N$, evaluation gives canonical isomorphisms $M\cong{}^*(M^*)$ and $N\cong({}^*N)^*$, sending $m$ to $(f\mapsto f(m))$ and $n$ to $(g\mapsto g(n))$, respectively.
For bimodules, both are bimodule isomorphisms.
No symmetrising form is needed.
\end{lemma}

\begin{proof}
By Theorem~\ref{prop:semisimple}, every finite-dimensional $S$-module is a direct summand of some $S^{\oplus n}$.
Evaluation is an isomorphism for $S$, hence for such summands; \eqref{eq:bimodule-dual-actions} gives bimodule compatibility.
\end{proof}

\section{Lecture 2: Tensor and quadratic algebras}

\subsection*{Tensor products over an algebra}

The degree zero algebra acts on both sides of the degree $1$ piece.
We therefore need the tensor product over an algebra, not only the tensor product over $\Bbbk$.

\begin{definition}
Let $M$ be a right $S$-module, let $N$ be a left $S$-module, and let $W$ be a $\Bbbk$-vector space.
The \defn{tensor product} $M\otimes_S N$ is the quotient of $M\otimes_\Bbbk N$ by the subspace generated by
\[
ms\otimes n-m\otimes sn
\qquad (m\in M,\ n\in N,\ s\in S).
\]
Equivalently, a bilinear map $b:M\times N\to W$ factors uniquely through a linear map $M\otimes_S N\to W$ precisely when $b(ms,n)=b(m,sn)$ for all $m\in M$, $s\in S$, and $n\in N$.
\end{definition}

The equality $ms\otimes_S n=m\otimes_S sn$ allows a scalar from $S$ to pass between adjacent factors.
For an $S$-bimodule $V$, define
\[
V^{\otimes_S 0}:=S,\qquad
V^{\otimes_S i}:=
\underbrace{V\otimes_S\cdots\otimes_S V}_{i\text{ factors}}
\quad (i\geq 1).
\]

Tensor duality reverses the order of the factors.

\begin{proposition}\label{prop:tensor-duals}
Let $S$ be finite-dimensional semisimple and let $V,W$ be finite-dimensional $S$-bimodules.
There are natural bimodule isomorphisms
\begin{equation}\label{eq:tensor-duals}
\begin{aligned}
\theta_D &: D(W)\otimes_S V^*\xrightarrow{\sim}D(V\otimes_S W),\\
\theta_r &: W^*\otimes_S V^*\xrightarrow{\sim}(V\otimes_S W)^*,\\
\theta_\ell &: {}^*W\otimes_S {}^*V\xrightarrow{\sim}{}^*(V\otimes_S W),\\
\theta_{\mathrm{en}} &: W^*\otimes_S V^{*_{\mathrm{en}}}
\xrightarrow{\sim}(V\otimes_S W)^{*_{\mathrm{en}}},
\end{aligned}
\end{equation}
defined by
\begin{equation}\label{eq:tensor-evaluations}
\begin{aligned}
\theta_D(\lambda\otimes f)(v\otimes w)&=\lambda(f(v)w),\\
\theta_r(g\otimes f)(v\otimes w)&=g(f(v)w),\\
\theta_\ell(g\otimes f)(v\otimes w)&=f(vg(w)),\\
\theta_{\mathrm{en}}(g\otimes F)(v\otimes w)&=\sum_j g(a_jw)\otimes b_j^\op,
\quad\text{where }F(v)=\sum_j a_j\otimes b_j^\op.
\end{aligned}
\end{equation}
\end{proposition}

\begin{proof}
The formulas in \eqref{eq:tensor-evaluations} respect the tensor relations and bimodule actions.
The maps $\theta_D$ and $\theta_r$ are isomorphisms for $V=S$ as a right module, hence for finite direct sums and direct summands of $S$.
For $\theta_\ell$, apply the same argument to the left module $W$.
Choose a symmetrising form $\tau$ as in Definition~\ref{def:symmetrising-form}, and use $\gamma_V,\gamma_{V\otimes_S W}$ from \eqref{eq:en-dual-comparison}.
Then
\[
\gamma_{V\otimes_S W}\circ\theta_{\mathrm{en}}
=\theta_r\circ(\id_{W^*}\otimes\gamma_V),
\]
since both sides evaluate $g\otimes F$ at $v\otimes w$ as $\sum_j g(a_jw)\tau(b_j)$.
Proposition~\ref{prop:dual-comparison} makes both $\gamma$ maps isomorphisms, so $\theta_{\mathrm{en}}$ is an isomorphism.
\end{proof}

For a fixed symmetrising form $\tau$, let $\alpha_M(f)=\tau\circ f$ as in \eqref{eq:dual-comparison}.
Then tensor evaluation satisfies
\[
\alpha_{V\otimes_S W}\circ\theta_r
=\theta_D\circ(\alpha_W\otimes\id_{V^*}).
\]
Both sides send $g\otimes f$ to $v\otimes w\mapsto\tau(g(f(v)w))$.

\begin{example}
Let $S=\Bbbk^{\oplus r}$ with coordinate idempotents $e_1,\ldots,e_r$, and suppose $V=e_iVe_j$ and $W=e_jWe_\ell$.
By \eqref{eq:bimodule-dual-actions},
\[
V\otimes_S W=e_i(V\otimes_S W)e_\ell,\qquad
W^*\otimes_S V^*=e_\ell(W^*\otimes_S V^*)e_i.
\]
Thus $\theta_r$ in \eqref{eq:tensor-duals} reverses the two outer idempotents.
\end{example}

\subsection*{Generated algebras}

Let $A$ be graded with $S=A_0$.
Multiplication makes $A_1$ an $S$-bimodule and gives
\begin{equation}\label{eq:multiplication-maps}
\mu_i:A_1^{\otimes_S i}\longrightarrow A_i,\qquad
a_1\otimes\cdots\otimes a_i\longmapsto a_1\cdots a_i
\quad(i\geq 1),\qquad \mu_0=\id_S.
\end{equation}

\begin{definition}
The algebra $A$ is \defn{generated in degree $1$ over $S$} if every $\mu_i$ in \eqref{eq:multiplication-maps} is surjective.
\end{definition}

Equivalently, $S$ and $A_1$ generate $A$ as an algebra.

\begin{example}
The algebra $\Bbbk A_n$ of Example~\ref{ex:triangular-grading} is generated in degree $1$ over $S=\bigoplus_i\Bbbk e_{ii}$:
\[
e_{ij}=e_{i,i+1}e_{i+1,i+2}\cdots e_{j-1,j}
\quad(1\leq i<j\leq n).
\]
For the grading $\deg e_{ij}=2(j-i)$, the same algebra has $A_1=0$ and $A_2\neq 0$, so it is not generated in degree $1$.
\end{example}

\subsection*{Tensor algebras and the universal property}

\begin{definition}\label{def:tensor-algebra}
Let $S$ be a $\Bbbk$-algebra and $V$ an $S$-bimodule.
The \defn{tensor algebra of $V$ over $S$} is
\[
T_S(V):=\bigoplus_{i\in\Z}T_S(V)_i,
\quad\text{where}\quad
T_S(V)_i=
\begin{cases}
V^{\otimes_S i},&i\geq 1,\\
S,&i=0,\\
0,&i<0,
\end{cases}
\]
with multiplication given by concatenation of tensors and the given $S$-actions.
\end{definition}

The left and right $S$-actions determine which factors can be concatenated.
For $S=\bigoplus_{i=1}^r\Bbbk e_i$ with coordinate idempotents,
\begin{equation}\label{eq:composable-tensors}
V=\bigoplus_{i,j}e_iVe_j,\qquad
(e_iVe_j)\otimes_S(e_\ell Ve_m)=0
\quad\text{if }j\neq\ell.
\end{equation}
Indeed, $v\otimes w=ve_j\otimes w=v\otimes e_jw=0$ when $v\in e_iVe_j$, $w\in e_\ell Ve_m$, and $j\neq\ell$.

\begin{example}\label{ex:tensor-A3}
Let $S=\Bbbk^{\oplus 3}$ with coordinate idempotents $e_1,e_2,e_3$, and let $V=\Bbbk a\oplus\Bbbk b$ have actions $e_1ae_2=a$ and $e_2be_3=b$, with all other idempotent actions zero.
By \eqref{eq:composable-tensors},
\[
V\otimes_S V=\Bbbk(a\otimes b),\qquad
V^{\otimes_S 3}=0,\qquad
T_S(V)=S\oplus\Bbbk a\oplus\Bbbk b\oplus\Bbbk ab.
\]
Here $ab=a\otimes b$.
The assignments $e_i\mapsto e_{ii}$, $a\mapsto e_{12}$, and $b\mapsto e_{23}$ give $T_S(V)\cong\Bbbk A_3$; the tensor $ab$ maps to $e_{13}$.
\end{example}

\begin{proposition}[Universal property of the tensor algebra]\label{prop:tensor-universal}
Let $\iota_S:S\to T_S(V)$ and $\iota_V:V\to T_S(V)$ be the canonical inclusions.
Given a unital $\Bbbk$-algebra homomorphism $\varphi:S\to B$ and an $S$-bimodule map $f:V\to B$, where $sbt=\varphi(s)b\varphi(t)$ on $B$, there is a unique $\Bbbk$-algebra homomorphism $\widetilde f:T_S(V)\to B$ making the diagram commute:
\begin{equation}\label{eq:tensor-universal}
\xymatrix@C=42pt@R=28pt{
V \ar[r]^{\iota_V} \ar[dr]_{f}
& T_S(V) \ar@{-->}[d]^{\exists!\,\widetilde f}
& S \ar[l]_{\iota_S} \ar[dl]^{\varphi}\\
& B. &
}
\end{equation}
\end{proposition}

\begin{proof}
Define $\widetilde f$ linearly by
\begin{equation}\label{eq:tensor-extension}
\widetilde f(s)=\varphi(s),\qquad
\widetilde f(v_1\otimes_S\cdots\otimes_S v_i)=f(v_1)\cdots f(v_i)
\quad(s\in S,\ i\geq 1).
\end{equation}
The identity $f(svt)=\varphi(s)f(v)\varphi(t)$ makes $\widetilde f$ well-defined on tensor products and compatible with multiplication.
Every homomorphism satisfying \eqref{eq:tensor-universal} has the values in \eqref{eq:tensor-extension}, proving uniqueness.
\end{proof}

\begin{corollary}\label{cor:tensor-presentation}
If $A$ is generated in degree $1$ over $S=A_0$, there is a surjective graded algebra homomorphism $\mu:T_S(A_1)\onto A$ whose degree $i$ component is $\mu_i$ from \eqref{eq:multiplication-maps}.
\end{corollary}

\begin{proof}
Apply Proposition~\ref{prop:tensor-universal} to the inclusions $S\to A$ and $A_1\to A$.
Surjectivity of each $\mu_i$ gives surjectivity of $\mu$.
\end{proof}

\begin{definition}\label{def:relations}
For the map $\mu$ in Corollary~\ref{cor:tensor-presentation}, the homogeneous two-sided ideal $I:=\Ker\mu$ is the \defn{ideal of relations}.
\end{definition}

Writing $V=A_1$, we obtain $A\cong T_S(V)/I$ with $I_0=I_1=0$.
The bimodule $V$ specifies the generators and their left and right $S$-actions; $I$ specifies the relations between their products.

\subsection*{Quadratic algebras}

\begin{definition}\label{def:quadratic}
Let $S$ be finite-dimensional semisimple, $V$ a finite-dimensional $S$-bimodule, and $R\subset V\otimes_S V$ an $S$-sub-bimodule.
The algebra $T_S(V)/(R)$ is a \defn{quadratic algebra}, where $(R)$ denotes the two-sided ideal generated by $R$.
\end{definition}

\begin{proposition}\label{prop:quadratic-basic}
A quadratic algebra $A=T_S(V)/(R)$ has $A_0=S$ and $A_1=V$, is locally finite, and is generated in degree $1$ over $S$.
\end{proposition}

\begin{proof}
The ideal $(R)$ has no components in degrees $0$ and $1$.
Each $A_i$ is a quotient of the finite-dimensional space $V^{\otimes_S i}$, and multiplication maps $V^{\otimes_S i}$ onto $A_i$.
\end{proof}

\begin{example}\label{ex:tensor-An}
Let $S=\bigoplus_{i=1}^n\Bbbk e_{ii}$ and $V=\bigoplus_{i=1}^{n-1}\Bbbk e_{i,i+1}$ inside $\Bbbk A_n$.
By \eqref{eq:composable-tensors}, the basis tensors of positive length are
\[
e_{i,i+1}\otimes_S e_{i+1,i+2}\otimes_S\cdots\otimes_S e_{j-1,j}
\longmapsto e_{ij}
\quad(1\leq i<j\leq n).
\]
Thus multiplication gives $T_S(V)\cong\Bbbk A_n$, a quadratic algebra with $R=0$.
\end{example}

\begin{example}\label{ex:zero-composition}
For $S,V,a,b$ from Example~\ref{ex:tensor-A3}, the quotient
\[
T_S(V)/(ab)\cong\Bbbk A_3/(e_{13})
\]
is quadratic with $R=\Bbbk(a\otimes b)$.
Its basis is $e_1,e_2,e_3,a,b$, in degrees $0,0,0,1,1$, as in Example~\ref{ex:quadratic-matrix-quotient}.
\end{example}

\begin{example}\label{ex:two-vertex-relations}
Let $S=\Bbbk^{\oplus 2}$ with coordinate idempotents $e_1,e_2$, and let $V=\Bbbk a\oplus\Bbbk b$ satisfy $e_1ae_2=a$ and $e_2be_1=b$, with all other idempotent actions zero.
In $T_S(V)$ only alternating words in $a,b$ survive.
The quadratic algebra $A=T_S(V)/(ab)$ has
\[
A=S\oplus\Bbbk a\oplus\Bbbk b\oplus\Bbbk ba,
\qquad \dim_\Bbbk A_0=2,\quad \dim_\Bbbk A_1=2,\quad \dim_\Bbbk A_2=1.
\]
Every alternating word of length at least $3$ contains $ab$, so all higher components vanish.
\end{example}

\begin{example}\label{ex:cubic-quotient}
Let $B=\Bbbk A_4$ and $A=B/(e_{14})$, with the grading of Example~\ref{ex:triangular-grading}.
The ideal $I=(e_{14})=\Bbbk e_{14}$ has $I_2=0$ but $I_3\neq 0$.
Since $e_{14}=e_{12}e_{23}e_{34}$, the algebra $A$ is generated in degree $1$ with a relation of degree $3$ and no relations of degree $2$.
Hence $A$ is not quadratic.
\end{example}

\begin{example}[Polynomial and exterior algebras]\label{ex:polynomial-exterior}
For the classical comparison, take $S=\Bbbk$ and $V=\bigoplus_{i=1}^n\Bbbk x_i$.
The \defn{symmetric algebra} and \defn{exterior algebra} are
\begin{equation}\label{eq:polynomial-exterior}
\begin{aligned}
\Symm(V)&=T_\Bbbk(V)/(x_ix_j-x_jx_i\mid i<j)
            \cong\Bbbk[x_1,\ldots,x_n],\\
\bigwedge V&=T_\Bbbk(V)/(v^2\mid v\in V).
\end{aligned}
\end{equation}
Both are quadratic.
The polynomial basis allows repetitions; the exterior basis consists of $1$ and $x_{i_1}\cdots x_{i_d}$ with $i_1<\cdots<i_d$.
The exterior relations are equivalently $x_i^2=0$ for $1\leq i\leq n$ and $x_ix_j+x_jx_i=0$ for $i<j$, in every characteristic.
\end{example}

\subsection*{Hilbert series}

A Hilbert series records the dimension of each homogeneous component.

\begin{definition}\label{def:hilbert-series}
Let $M=\bigoplus_{j\in\Z}M_j$ be a graded vector space with $\dim_\Bbbk M_j<\infty$ for all $j$ and $M_j=0$ for all $j\ll 0$.
Its \defn{Hilbert series} is the formal Laurent series $h_M(t):=\sum_{j\in\Z}(\dim_\Bbbk M_j)t^j$.
\end{definition}

For a locally finite graded algebra $A$, $h_A(t)$ refers to its underlying graded vector space; the series does not record multiplication.

\begin{lemma}\label{lem:hilbert-operations}
Let $M,N$ be locally finite graded vector spaces concentrated in degrees bounded below.
\begin{enumerate}
\item A graded isomorphism $M\cong N$ implies $h_M(t)=h_N(t)$.
\item An exact sequence $0\to L\to M\to N\to0$ with degree $0$ maps gives $h_M(t)=h_L(t)+h_N(t)$.
\item For every $d\in\Z$, $h_{M(d)}(t)=t^{-d}h_M(t)$.
\end{enumerate}
\end{lemma}

\begin{proof}
The first two statements follow from dimensions in each degree. For shifts,
\[
h_{M(d)}(t)=\sum_j\dim_\Bbbk(M_{j+d})t^j=t^{-d}\sum_i\dim_\Bbbk(M_i)t^i.\qedhere
\]
\end{proof}

\begin{example}
For the upper triangular algebra in Example~\ref{ex:triangular-grading}, counting matrix entries on each diagonal gives
\[
h_{\Bbbk A_n}(t)=n+(n-1)t+\cdots+t^{n-1},\qquad
h_{\Bbbk A_3/(e_{13})}(t)=3+2t.
\]
The second equality also follows by subtracting $h_{(e_{13})}(t)=t^2$ from $h_{\Bbbk A_3}(t)$, as in Lemma~\ref{lem:hilbert-operations}(2).
For $A=T_S(V)/(ab)$ in Example~\ref{ex:two-vertex-relations}, the basis $e_1,e_2,a,b,ba$ gives $h_A(t)=2+2t+t^2$.
\end{example}

\begin{example}
For the algebras in Example~\ref{ex:polynomial-exterior}, each polynomial generator may occur any number of times, whereas each exterior generator occurs at most once. Thus
\[
h_{\Symm(V)}(t)=\frac{1}{(1-t)^n},\qquad
h_{\bigwedge V}(t)=(1+t)^n
\quad\text{where }n=\dim_\Bbbk V.
\]
\end{example}

\begin{lemma}\label{lem:hilbert-tensor}
For locally finite graded vector spaces $V,W$ concentrated in degrees bounded below,
\[
h_{V\otimes_\Bbbk W}(t)=h_V(t)h_W(t),\qquad
(V\otimes_\Bbbk W)_n=\bigoplus_{i+j=n}V_i\otimes_\Bbbk W_j.
\]
\end{lemma}

\begin{proof}
The coefficient of $t^n$ on either side is $\sum_{i+j=n}\dim_\Bbbk V_i\dim_\Bbbk W_j$. Only finitely many summands are non-zero because both gradings are bounded below.
\end{proof}

\subsection*{The degree 2 relations of a quotient}

Let $S$ be finite-dimensional semisimple and $V$ a finite-dimensional $S$-bimodule.
Let $A=T_S(V)/I$ with $I_0=I_1=0$ as in Definition~\ref{def:relations}.
In degree $2$,
\[
A_2\cong(V\otimes_S V)/I_2,
\quad\text{where}\quad
I_2=I\cap(V\otimes_S V).
\]

\begin{definition}\label{def:quadratic-part}
The \defn{quadratic part} of $A$ is $qA:=T_S(V)/(I_2)$.
\end{definition}

The inclusion $(I_2)\subset I$ induces a surjective graded homomorphism $\pi_A:qA\onto A$, and
\[
\pi_A\text{ is an isomorphism}
\quad\Longleftrightarrow\quad I=(I_2)
\quad\Longleftrightarrow\quad A\text{ is quadratic}.
\]

\begin{example}
For $A=\Bbbk A_4/(e_{14})$ from Example~\ref{ex:cubic-quotient}, we have $I_2=0$, so
\[
qA=\Bbbk A_4,\qquad
\pi_A:\Bbbk A_4\onto\Bbbk A_4/(e_{14}),\qquad
\Ker\pi_A=\Bbbk e_{14}.
\]
Thus $qA$ does not impose the cubic relation $e_{12}e_{23}e_{34}=0$.
\end{example}

\subsection*{Orthogonal complements and quadratic duality}

\begin{definition}\label{def:orthogonals}
For an $S$-sub-bimodule $U$ of a finite-dimensional $S$-bimodule $M$, define the \defn{right orthogonal complement} and \defn{left orthogonal complement} by
\[
U^\perp:=\{f\in M^*\mid f(U)=0\},
\qquad
{}^\perp U:=\{g\in{}^*M\mid g(U)=0\}.
\]
\end{definition}

Both $U^\perp$ and $ {}^\perp U$ are $S$-sub-bimodules by \eqref{eq:bimodule-dual-actions}.
Fix a symmetrising form $\tau$ as in Definition~\ref{def:symmetrising-form}.
For the comparison maps $\alpha_M(f)=\tau\circ f$ and $\beta_M(g)=\tau\circ g$ in \eqref{eq:dual-comparison},
\begin{equation}\label{eq:orthogonal-comparison}
\alpha_M(U^\perp)=\beta_M({}^\perp U)
=\{\lambda\in D(M)\mid\lambda(U)=0\}.
\end{equation}
Indeed, if $\alpha_M(f)(U)=0$, then
\[
\tau(f(u)s)=\tau(f(us))=0
\quad\text{for all }s\in S,\ u\in U,
\]
so non-degeneracy of $\tau$ gives $f(U)=0$.
For $\beta_M$, use $\tau(sg(u))=\tau(g(su))$ instead.

For $A=T_S(V)/(R)$, the map $\theta_r$ in \eqref{eq:tensor-duals} identifies $R^\perp$ with a sub-bimodule of $V^*\otimes_S V^*$.
These are the generators and relations used for the quadratic dual in Lecture~3:
\[
V^*=\Hom_S(V,S),\qquad
R^\perp\subset V^*\otimes_S V^*.
\]
The generators $V^*$ are assigned degree $1$.
In contrast, the graded $S$-dual $V^{(*)}$ from Definition~\ref{def:graded-S-duals} has degree $-1$ when $V=A_1$.

\begin{example}
For $\Symm(V)$ and $\bigwedge V$ in Example~\ref{ex:polynomial-exterior}, take $V=\Bbbk x\oplus\Bbbk y$.
The relation spaces are
\[
\begin{aligned}
R_{\Symm(V)}&=\Bbbk(x\otimes y-y\otimes x),\\
R_{\bigwedge V}&=\Bbbk(x\otimes x)\oplus\Bbbk(y\otimes y)\oplus\Bbbk(x\otimes y+y\otimes x).
\end{aligned}
\]
Let $x^*,y^*\in V^*$ be the coordinate functionals.
With the tensor evaluation $\theta_r$ in \eqref{eq:tensor-evaluations},
\[
\begin{aligned}
R_{\Symm(V)}^\perp&=\operatorname{span}_\Bbbk\{x^*\otimes x^*,\,y^*\otimes y^*,\,x^*\otimes y^*+y^*\otimes x^*\},\\
R_{\bigwedge V}^\perp&=\Bbbk(x^*\otimes y^*-y^*\otimes x^*).
\end{aligned}
\]
Thus the quadratic dual presentations exchange the symmetric and exterior relations.
\end{example}
